function averagePresent(values){
  const nums = values.map(function(v){
    if(v===null||v===undefined||String(v).trim()==='') return NaN;
    return Number(v);
  }).filter(function(n){ return Number.isFinite(n); });
  return nums.length ? Math.round(nums.reduce(function(a,b){return a+b;},0) / nums.length) : '';
}
function validScore(v){
  const t = toWesternDigits(v == null ? '' : v).replace(/[^\d.]/g,'');
  if(t === '') return '';
  const n = Math.max(0,Math.min(100,Number(t)));
  return Number.isFinite(n) ? n : '';
}
function gradeWords(v){ const n=validScore(v); return n === '' ? '' : numberToArabicWords(n); }
function gbAllScores(values){
  const nums=[];
  for(let i=0;i<values.length;i++){
    const n=validScore(values[i]);
    if(n==='')return '';
    nums.push(Number(n));
  }
  return nums.length?Math.round(nums.reduce(function(a,b){return a+b;},0)/nums.length):'';
}
function gbBestTwoAvg(values){
  const nums=values.map(function(v){if(v===null||v===undefined||String(v).trim()==='')return NaN;const n=Number(typeof validScore==='function'?validScore(v):v);return Number.isFinite(n)?n:NaN;}).filter(function(n){return Number.isFinite(n);});
  if(!nums.length)return '';
  if(nums.length<=2)return Math.round(nums.reduce(function(a,b){return a+b;},0)/nums.length);
  nums.sort(function(a,b){return b-a;});
  return Math.round((nums[0]+nums[1])/2);
}
function gbPrepareMonthlyRow(row){
  if(!row||row._gradebookMonthlyV2)return row;
  // Keep the previous fields intact. Copy the old upper-grade values into the
  // closest new fields only once, so old records remain readable and editable.
  if(row.t1m1==null||row.t1m1==='')row.t1m1=row.oct==null?'':row.oct;
  if(row.t1m1Daily==null||row.t1m1Daily==='')row.t1m1Daily=row.nov==null?'':row.nov;
  if(row.t2m1==null||row.t2m1==='')row.t2m1=row.feb==null?'':row.feb;
  if(row.t2m1Daily==null||row.t2m1Daily==='')row.t2m1Daily=row.apr==null?'':row.apr;
  if(row.result==null||row.result==='')row.result=row.finalGrade!=null&&row.finalGrade!==''?row.finalGrade:(row.finalExam==null?'':row.finalExam);
  row._gradebookMonthlyV2=true;
  return row;
}
function gbMonthlyAverage(exam,daily){
  const a=validScore(exam),b=validScore(daily),hasA=a!==''&&a!=null,hasB=b!==''&&b!=null;
  if(hasA&&hasB)return Math.round((Number(a)+Number(b))/2);
  if(hasA)return Number(a);
  if(hasB)return Number(b);
  return '';
}
function gbCompute56(row,recOrBest){
  if(!row)return;
  if(row._gradebookMonthlyV2){
    row.t1m1Avg=gbMonthlyAverage(row.t1m1,row.t1m1Daily);
    row.t1m2Avg=gbMonthlyAverage(row.t1m2,row.t1m2Daily);
    row.t2m1Avg=gbMonthlyAverage(row.t2m1,row.t2m1Daily);    row.t2m2Avg=gbMonthlyAverage(row.t2m2,row.t2m2Daily);
    const best=recOrBest===true||(recOrBest&&recOrBest.term1BestTwo);
    row.term1=best?gbBestTwoAvg([row.t1m1Avg,row.t1m2Avg]):averagePresent([row.t1m1Avg,row.t1m2Avg]);
    row.term2=averagePresent([row.t2m1Avg,row.t2m2Avg]);
    row.coursework=averagePresent([row.term1,validScore(row.mid),row.term2]);
    const result=validScore(row.result);
    if(result!==''&&result!=null)row.finalGrade=Number(result);
    else if(row.finalGrade==null)row.finalGrade='';
    return;
  }
  const best=recOrBest===true||(recOrBest&&recOrBest.term1BestTwo);
  row.term1 = best ? gbBestTwoAvg([row.oct,row.nov,row.dec]) : averagePresent([row.oct,row.nov,row.dec]);
  row.term2 = gbAllScores([row.feb,row.apr]);
  row.coursework = (row.term1!==''&&validScore(row.mid)!==''&&row.term2!=='') ? gbAllScores([row.term1,row.mid,row.term2]) : '';
  row.finalGrade = (row.coursework!==''&&validScore(row.finalExam)!=='') ? Math.round((Number(row.coursework)+Number(validScore(row.finalExam)))/2) : '';
}

function seedDemoGradesForSection(grade,section){
  if(!section||!Array.isArray(section.students)) return;
  const subjects=gbSubjects(grade);
  section.students.forEach(function(student,studentIndex){
    if(!student.demoDefault&&!/^st_demo_/.test(student.id||'')&&!/الطالب (الأول|الثاني|الثالث)/.test(String(student.name||''))) return;
    const resultType=demoStudentResultType(student,studentIndex);
    subjects.forEach(function(subject,subjectIndex){
      const record=gbRecord(grade,section.id,subject,true);
      const row=record.rows[student.id]||(record.rows[student.id]={});
      const base=resultType==='fail'?(38+(subjectIndex%5)):(studentIndex%3===2?68+(subjectIndex%6):82+(subjectIndex%7));
      if(usesSixthGradeLayout(grade)){
        if(row.oct==null||row.oct==='') row.oct=Math.min(100,base);
        if(row.nov==null||row.nov==='') row.nov=Math.min(100,base+2);
        if(row.dec==null||row.dec==='') row.dec=Math.max(0,base-1);
        if(row.mid==null||row.mid==='') row.mid=Math.min(100,base+1);
        if(row.feb==null||row.feb==='') row.feb=Math.min(100,base+3);
        if(row.apr==null||row.apr==='') row.apr=Math.min(100,base+2);
        if(row.finalExam==null||row.finalExam==='') row.finalExam=resultType==='fail'?Math.min(48,base+2):Math.min(100,base+4);
        gbCompute56(row);
      }else{
        if(row.mid==null||row.mid==='') row.mid=resultType==='fail'?Math.min(47,base+2):Math.min(100,base+2);
        if(row.finalExam==null||row.finalExam==='') row.finalExam=resultType==='fail'?Math.min(46,base+1):Math.min(100,base+5);
        if(row.remarks==null||row.remarks==='') row.remarks=resultType==='fail'?'يحتاج إلى متابعة':'ناجح';
      }
    });
  });
}

function gradebookDecisionNumber(value){

function formatDecisionPoints(value){
  const n=Number(value)||0;
  return Number.isInteger(n)?String(n):String(Number(n.toFixed(2)));
}
function gradebookDecisionAllocations(grade,section,record){
  const allocations={};
  if(!section||!record)return allocations;
  let remaining=gradebookDecisionNumber(record.decisionPoints);
  if(remaining<=0)return allocations;
  const upper=usesSixthGradeLayout(grade), candidates=[];
  (section.students||[]).forEach(function(st,index){
    const row=record.rows&&record.rows[st.id];if(!row)return;
    if(upper)gbCompute56(row,record);
    const raw=gradebookNumeric(upper?row.finalGrade:row.finalExam);
    if(raw!=null&&raw<50){
      candidates.push({id:st.id,raw:raw,need:50-raw,index:index});
    }
  });
  // Distribute the available decision points to failing grades only,
  // starting with the grade nearest to 50. A partial allocation is valid:
  // for example, 10 points on a grade of 39 produces 49 rather than doing
  // nothing because 11 points would be needed to reach 50.
  candidates.sort(function(a,b){return a.need-b.need||a.index-b.index;});
  candidates.forEach(function(item){
    if(remaining<=0)return;
    const applied=Math.min(remaining,item.need);
    if(applied>0){
      const effective=Number((item.raw+applied).toFixed(2));
      allocations[item.id]={raw:item.raw,applied:applied,effective:effective};
      remaining-=applied;
    }
  });
  Object.keys(allocations).forEach(function(id){allocations[id].remainingAfter=Math.max(0,remaining);});
  allocations.__remaining=remaining;
  return allocations;
}

function gradebookNumeric(v){
  if(v===''||v==null) return null;
  const n=Number(toWesternDigits(v));
  return Number.isFinite(n)?n:null;
}
function updateGradebookSummary(){
  const gradeEl=document.getElementById('gbGrade'), secEl=document.getElementById('gbSection'), subEl=document.getElementById('gbSubject');
  const foot=document.getElementById('gbTableFoot'), card=document.getElementById('gbSectionPassCard');
  if(!gradeEl||!secEl||!subEl||!foot) return;
  const grade=gradeEl.value, sectionId=secEl.value, subject=subEl.value;
  const section=gbSectionById(grade,sectionId);
  const record=gbRecord(grade,sectionId,subject,false);
  if(!section||!record){foot.innerHTML=''; if(card) card.innerHTML=''; return;}
  const upper=usesSixthGradeLayout(grade);
  const fields=upper?['oct','nov','dec','term1','mid','feb','apr','term2','coursework','finalExam','finalGrade']:['mid','finalExam'];
  const sums={}, counts={};
  fields.forEach(function(f){sums[f]=0;counts[f]=0;});
  let pass=0, considered=0;
  (section.students||[]).forEach(function(st){
    const r=record.rows[st.id]||{};
    if(upper) gbCompute56(r);
    fields.forEach(function(f){
      const n=gradebookNumeric(r[f]);
      if(n!=null){sums[f]+=n;counts[f]++;}
    });
    const fin=gradebookNumeric(typeof window.nbGbOfficialScore==='function'?window.nbGbOfficialScore(grade,sectionId,subject,st.id):(upper?r.finalGrade:r.finalExam));
    const mark=resultPassMark(grade);
    if(fin!=null){considered++; if(fin>=mark) pass++;}
  });
  const full=resultFullMark(grade);
  const totalStu=(section.students||[]).filter(function(st){return String(st.name||'').trim();}).length;
  function cell(f,kind){
    if(!counts[f]) return '';
    if(kind==='sum') return showGrade(Math.round(sums[f]));
    const pct=Math.round(sums[f]/(counts[f]*full)*100);
    return toArabicDigits(String(pct))+'٪';
  }
  let sumCells='', pctCells='';
  if(upper){
    ['oct','nov','dec','term1','mid','feb','apr','term2','coursework','finalExam','finalGrade'].forEach(function(f){
      sumCells+='<td>'+cell(f,'sum')+'</td>';
      pctCells+='<td>'+cell(f,'pct')+'</td>';
    });
    foot.innerHTML='<tr class="gb-sum-row"><td></td><td class="gb-sum-label">المجموع</td>'+sumCells+'</tr><tr class="gb-pct-row"><td></td><td class="gb-sum-label">النسبة المئوية</td>'+pctCells+'</tr>';
  }else{
    foot.innerHTML='<tr class="gb-sum-row"><td></td><td class="gb-sum-label">المجموع</td><td>'+cell('mid','sum')+'</td><td></td><td>'+cell('finalExam','sum')+'</td><td></td><td></td></tr>'+
      '<tr class="gb-pct-row"><td></td><td class="gb-sum-label">النسبة المئوية</td><td>'+cell('mid','pct')+'</td><td></td><td>'+cell('finalExam','pct')+'</td><td></td><td></td></tr>';
  }
  const rate=totalStu?Math.round((pass*100)/totalStu):0;
  const rateTxt=totalStu?toArabicDigits(String(rate))+'٪':'—';
  if(card) card.innerHTML='<b>نسبة نجاح الشعبة</b><strong>'+rateTxt+'</strong>';
  return {footHtml:foot.innerHTML, passHtml:'<div class="gb-print-pass"><b>نسبة نجاح الشعبة</b><strong>'+rateTxt+'</strong></div>', rate:rateTxt};
}


function computeSectionSubjectRates(grade,sectionId){
  const section=gbSectionById(grade,sectionId);
  const subjects=gbSubjects(grade);
  const students=(section&&section.students||[]).filter(function(st){return String(st.name||'').trim();});
  const total=students.length, mark=resultPassMark(grade), upper=usesSixthGradeLayout(grade);
  const rows=subjects.map(function(subject){
    let pass=0;
    students.forEach(function(st){
      const record=gbRecord(grade,sectionId,subject,false);
      const row=record&&record.rows?record.rows[st.id]:null;
      if(!row) return;
      if(upper) gbCompute56(row);
      const fin=gradebookNumeric(typeof window.nbGbOfficialScore==='function'?window.nbGbOfficialScore(grade,sectionId,subject,st.id):(upper?row.finalGrade:row.finalExam));
      if(fin!=null&&fin>=mark) pass++;
    });
    const pct=total?Math.round((pass*100)/total):0;
    return {subject:subject,pass:pass,total:total,pct:pct};
  });
  let overallPass=0;
  students.forEach(function(st){
    if(studentOfficialResult(grade,sectionId,st.id).key==='passed') overallPass++;
  });
  const overall=total?Math.round((overallPass*100)/total):0;
  return {section:section,total:total,rows:rows,overall:overall,overallPass:overallPass};
}

function markGradebookFailCells(){
  const gradeEl=document.getElementById('gbGrade');

  document.querySelectorAll('#gbTableHost tbody tr[data-student]').forEach(function(tr){
    tr.querySelectorAll('.grade-input').forEach(function(inp){
      const n=gradebookNumeric(inp.value);
      const fail=n!=null&&n<mark;
      inp.classList.toggle('gb-fail', fail);

      td.classList.toggle('gb-fail', n!=null&&n<mark);
    });
  });

function splitRowsFillingPages(doc,head,foot,theadHtml,rowHtmls,tableClass){
  const n=rowHtmls.length;
  if(!n) return [schoolSheetPageHtml(head,foot,theadHtml,'',tableClass)];
  doc.body.innerHTML=schoolSheetPageHtml(head,foot,theadHtml,rowHtmls[0],tableClass);
  const page=doc.querySelector('.exam-list-page');
  const hd=doc.querySelector('.exam-list-head');
  const ft=doc.querySelector('.exam-list-footer');
  const thead=doc.querySelector('thead');
  const tr=doc.querySelector('tbody tr');
  let pageH=0;
  try{ pageH=(doc.defaultView&&doc.defaultView.innerHeight)||0; }catch(e){}
  if(!pageH && page) pageH=page.offsetHeight||page.getBoundingClientRect().height||0;
  if(!pageH) pageH=740;
  const hh=hd?(hd.offsetHeight||90):90;
  const fh=45;
  const th=thead?(thead.offsetHeight||40):40;
  const rh=Math.max(tr?(tr.offsetHeight||0):0, 26);
  const per=Math.max(5, Math.floor((pageH-hh-fh-th-20)/rh));
  const pages=[];
  let i=0;
  const reserve=(typeof arguments[6]==='number'?arguments[6]:0);
  while(i<n){
    const left=n-i;
    const last=left<=per;
    let take=per;
    if(last && reserve) take=Math.max(1, per-reserve);
    if(take>left) take=left;
    pages.push(schoolSheetPageHtml(head,foot,theadHtml,rowHtmls.slice(i,i+take).join(''),tableClass));
    i+=take;
  }
  return pages;
}

function fillResultCardFromGradebook(card,grade,sectionId,studentId){
  if(!card) return;
  const rows=Array.from(card.querySelectorAll('table tbody tr'));
  const totals=[];
  let hasGrade=false,hasFail=false;
  rows.forEach(function(tr){
    const subjectEl=tr.querySelector('td.subject');
    const subject=subjectEl?subjectEl.textContent.trim():'';
    const boxes=Array.from(tr.querySelectorAll('.grade-box'));
    if(!subject||subject==='المجموع'||!boxes.length) return;
    const record=gbRecord(grade,sectionId,subject,false);
    const row=record&&record.rows?record.rows[studentId]:null;
    if(!row) return;
    if(usesSixthGradeLayout(grade)) gbCompute56(row);
    let values=[];
    if(card.classList.contains('secondary')){
      values=[row.term1,row.mid,row.term2,row.coursework,row.finalExam,row.finalGrade,'',''];
    }else if(card.classList.contains('upper')){
      values=[row.oct,row.nov,row.term1,row.mid,row.feb,row.apr,row.term2,row.coursework,row.finalExam,row.finalGrade];
    }else{
      const mid=validScore(row.mid),fin=validScore(row.finalExam);
      if(boxes[0]) setResultGradeBox(boxes[0],mid);
      if(boxes[1]) boxes[1].textContent=gradeWords(mid);
      if(boxes[2]) setResultGradeBox(boxes[2],fin);
      if(boxes[3]) boxes[3].textContent=gradeWords(fin);
      values=[mid,'',fin,''];
    }
    if(!(!card.classList.contains('secondary')&&!card.classList.contains('upper'))){
      values.forEach(function(v,i){ if(i<boxes.length)setResultGradeBox(boxes[i],v); });
    }
    values.forEach(function(v,i){
      const n=Number(v);
      if(v!==''&&v!=null&&Number.isFinite(n)){totals[i]=(totals[i]||0)+n;hasGrade=true;}
    });
    const finalValue=validScore(row.finalGrade!==undefined&&row.finalGrade!==''?row.finalGrade:row.finalExam);
    if(finalValue!==''&&Number(finalValue)<resultPassMark(grade))hasFail=true;
  });
  const totalRow=rows.find(function(tr){const s=tr.querySelector('td.subject');return s&&s.textContent.trim()==='المجموع';});
  if(totalRow){
    const boxes=Array.from(totalRow.querySelectorAll('.grade-box'));
    if(!card.classList.contains('upper')){
      if(totals[0]!=null){setResultGradeBox(boxes[0],totals[0]);if(boxes[1])boxes[1].textContent=numberToArabicWords(totals[0]);}
      if(totals[2]!=null){setResultGradeBox(boxes[2],totals[2]);if(boxes[3])boxes[3].textContent=numberToArabicWords(totals[2]);}
    }else{
      totals.forEach(function(v,i){if(boxes[i]&&v!=null)setResultGradeBox(boxes[i],v);});
    }
  }
  const resultEl=card.querySelector('.box.result .val');
  if(resultEl&&hasGrade)resultEl.textContent=hasFail?'راسب':'ناجح';
  if(card.classList.contains('secondary')) bindSecondaryCalc(card);
  refreshCardQR(card);
}

function ensureExamStudentMeta(grade,section,student){

function ttParseHtmlTable(html){
  let d; try{ d=new DOMParser().parseFromString(String(html||''),'text/html').body; }catch(e){ return null; } if(!d) return null;
  const table=d.querySelector('table'); if(!table)return null;
  const trs=[].slice.call(table.querySelectorAll('tr'));
  const grid=[];
  trs.forEach(function(tr,r){    if(!grid[r]) grid[r]=[];
    let col=0;
    [].slice.call(tr.cells||tr.querySelectorAll('th,td')).forEach(function(cell){
      while(grid[r][col]!=null) col++;
      const rs=Math.max(1,parseInt(cell.getAttribute('rowspan')||'1',10)||1);
      const cs=Math.max(1,parseInt(cell.getAttribute('colspan')||'1',10)||1);
      const val=ttCellText(cell);
      for(let i=0;i<rs;i++){
        if(!grid[r+i]) grid[r+i]=[];
        for(let j=0;j<cs;j++){
          if(grid[r+i][col+j]==null) grid[r+i][col+j]=(j===0?val:'');
        }
      }
      col+=cs;
    });
  });
  const max=grid.reduce(function(m,row){return Math.max(m,(row||[]).length);},0);
  const rows=grid.map(function(row){
    const a=[]; for(let i=0;i<max;i++) a.push(row&&row[i]!=null?row[i]:'');
    return a;
  }).filter(function(row){return row.some(function(x){return String(x).trim();});});
  return rows.length?rows:null;
}
function ttParseText(text){
  text=String(text||'').replace(/\r/g,'');
  const lines=text.split('\n').map(function(l){return l.replace(/\u00a0/g,' ');});
  const rows=[];
  lines.forEach(function(l){
    if(l.indexOf('\t')>=0){
      rows.push(l.split('\t').map(function(x){return x.replace(/[ \t]+\n/g,'\n').trim();}));
      return;
    }
    if(!l.trim()) return;
    rows.push(l.split(/\s{2,}|\s*\|\s*/).map(function(x){return x.trim();}));
  });
  return rows;
}

function ttImportExcel(input){
  const f=input.files&&input.files[0]; input.value='';
  if(!f)return;
  const name=(f.name||'').toLowerCase();
  const reader=new FileReader();
  reader.onerror=function(){alert('تعذر فتح الملف من الجهاز.');};
  reader.onload=function(e){
    const buf=e.target.result;
    function fail(msg){alert(msg||'تعذر قراءة ملف الإكسل. جرّب: حفظه من إكسل بصيغة .xlsx أو انسخ الجدول والصقه.');}
    const asText=function(){
      try{
        const u8=buf instanceof ArrayBuffer?new Uint8Array(buf):null;
        let txt='';
        if(typeof buf==='string') txt=buf;
        else if(u8){
          try{txt=new TextDecoder('utf-8').decode(u8);}catch(er){txt='';}
          if(!txt||txt.indexOf('�')===0){
            try{txt=new TextDecoder('windows-1256').decode(u8);}catch(er2){}
          }
        }
        if(txt&&/<(table|html|Workbook|ss:Workbook)/i.test(txt)){
          const rows=ttParseHtmlTable(txt);
          if(rows&&rows.length){ttSetRows(rows);return true;}
        }
        if(txt&&(txt.indexOf('\t')>=0||txt.indexOf(',')>=0||txt.indexOf(';')>=0)){
          const rows=ttParseText(txt.replace(/;/g,'\t').replace(/,/g, function(m,i,s){
            return (s.split('\n')[0]||'').split(',').length>2?'\t':m;
          }));
          if(rows&&rows.length>=2){ttSetRows(rows);return true;}
        }
      }catch(er){}
      return false;
    };
    ttLoadXlsx(function(ok){
      if(ok){
        try{
          let wb=null;
          try{wb=XLSX.read(new Uint8Array(buf),{type:'array',cellDates:true,raw:false});}catch(e1){
            try{wb=XLSX.read(buf,{type:'array'});}catch(e2){
              try{wb=XLSX.read(buf,{type:'binary'});}catch(e3){wb=null;}
            }
          }
          if(wb&&wb.SheetNames&&wb.SheetNames.length){
            const sh=wb.Sheets[wb.SheetNames[0]];
            const data=XLSX.utils.sheet_to_json(sh,{header:1,raw:false,defval:''});
            const rows=ttRowsFromAoA(data);
            if(rows.length){ttSetRows(rows);return;}
          }
        }catch(err){}
      }
      if(asText()) return;
      fail(name.indexOf('.xls')>=0&&name.indexOf('.xlsx')<0
        ?'هذا الملف قد يكون بصيغة إكسل القديمة. افتحه في إكسل واحفظه «مصنف Excel (.xlsx)» أو انسخ الجدول والصقه.'
        :'تعذر قراءة الملف. انسخ الجدول من التطبيق والصقه في صندوق اللصق ثم اضغط «لصق إكسل».');
    });
  };
  reader.readAsArrayBuffer(f);
}

function ttMatchStudentCol(header, grade, secLabel){
  const h=String(header||'').replace(/\s+/g,'');
  if(!h||ttHeaderIsMeta(h)) return false;
  const n=String(ttGradeNum(grade));
  const L=ttSecLetter(secLabel);
  if(n && L && (h===n+L || h===L+n || h===n+'-'+L || h===n+'/'+L)) return true;
  if(n && h.indexOf(n)!==-1 && L && h.indexOf(L)!==-1) return true;
  const g=String(grade||'').replace(/\s+/g,'');
  if(g && h.indexOf(g)!==-1 && L && h.indexOf(L)!==-1) return true;
  return false;
}

function parentPhoneDigits(raw){
  let d=String(raw||'').replace(/\D+/g,'');
  if(!d)return '';
  if(d.indexOf('00')===0)d=d.slice(2);
  if(d.charAt(0)==='0' && d.length>=10) d='964'+d.slice(1);
  if(d.indexOf('964')!==0 && d.length===10 && d.charAt(0)==='7') d='964'+d;
  return d;
}

function attendanceCalendarCounts(){
  const start=attendanceFirstWorkDate();
  const today=new Date();today.setHours(12,0,0,0);
  const end=today<start?new Date(start.getTime()):today;
  const startKey=typeof localDateKey==='function'?localDateKey(start):start.toISOString().slice(0,10);
  const endKey=typeof localDateKey==='function'?localDateKey(end):end.toISOString().slice(0,10);
  const workSet={};
  (ATTENDANCE_STATE.records||[]).forEach(function(r){
    if(!r||!r.date)return;
    if(r.date<startKey||r.date>endKey)return;
    if(r.status==='غائب'||r.status==='مجاز'||r.status==='مجاز بعذر') workSet[r.date]=1;
  });
  const workDates=Object.keys(workSet).sort();
  const holidayDates=[];
  let total=0,d=new Date(start.getTime());
  while(d<=end){
    total++;
    const key=typeof localDateKey==='function'?localDateKey(d):d.toISOString().slice(0,10);
    if(!workSet[key]) holidayDates.push(key);
    d.setDate(d.getDate()+1);
  }
  return {work:workDates.length,holidays:holidayDates.length,workDates:workDates,holidayDates:holidayDates};
}

function ymdDiff(fromDate,toDate){if(!fromDate)return {years:0,months:0,days:0,empty:true};let from=new Date(fromDate+'T12:00:00'),to=toDate?new Date(toDate):new Date();if(isNaN(from))return {years:0,months:0,days:0,empty:true};if(to<from)return {years:0,months:0,days:0};let years=to.getFullYear()-from.getFullYear(),months=to.getMonth()-from.getMonth(),days=to.getDate()-from.getDate();if(days<0){months--;days+=new Date(to.getFullYear(),to.getMonth(),0).getDate();}if(months<0){years--;months+=12;}return {years:Math.max(0,years),months:months,days:days};}

function smartStaffColMap(headers){
  const map={};
  (headers||[]).forEach(function(c,i){
    const t=String(c||'').replace(/\s+/g,' ').trim();
    if(!t)return;
    if((/اسم/.test(t) && !/أم|ام|الأم|الام|مستمسك/.test(t) && map.name==null) || /^(الاسم|اسم|name|الاسم الكامل)$/i.test(t)) map.name=i;
    else if(/لقب|اللقب/i.test(t)) map.title=i;
    else if(/عنوان وظيف|وظيفة|مسمى وظيف|العنوان|المسمى|job|مسمى/i.test(t) && map.jobTitle==null) map.jobTitle=i;
    else if(/رقم.*(وظيف|موظف|تسلسل)|الرقم الوظيف|employee.*no/i.test(t)) map.employeeNo=i;
    else if(/ميلاد|مواليد|تولد|birth/i.test(t)) map.birthDate=i;
    else if(/أم$|اسم الأم|اسم الام|الأم|الام|mother/i.test(t) && !/أمر/.test(t)) map.motherName=i;
    else if(/تعيين|appointment/i.test(t) && map.appointmentDate==null) map.appointmentDate=i;
    else if(/مباشرة.*هذه|هذه.*مدر|school.*start/i.test(t)) map.schoolStartDate=i;
    else if(/مباشرة|باشر|start.*date/i.test(t) && map.startDate==null) map.startDate=i;
    else if(/أمر إدار|امر ادار|admin.*order/i.test(t)) map.adminOrderDate=i;
    else if(/اختصاص|تخصص|special/i.test(t)) map.specialization=i;
    else if(/هاتف|جوال|موبايل|phone|mobile/i.test(t)) map.phone=i;
    else if(/بطاقة|وطني|هوية|national/i.test(t)) map.nationalId=i;
    else if(/حالة|وضع|status/i.test(t)) map.status=i;
  });
  if(map.name==null){
    for(var i=0;i<(headers||[]).length;i++){
      var t=String(headers[i]||'').trim();
      if(t && /[\u0600-\u06FF]/.test(t) && !/ت|م|رقم|تاريخ|هاتف/.test(t)){map.name=i;break;}
    }
  }
  return map;
}
function ingestStaffGrid(rows, kind){
  if(!rows||!rows.length) return 0;
  var headerRow=0, bestScore=0;
  for(var r=0;r<Math.min(5,rows.length);r++){
    var score=0;
    (rows[r]||[]).forEach(function(c){var t=String(c||'').trim();if(t && /[\u0600-\u06FFa-zA-Z]/.test(t))score++;});
    if(score>bestScore){bestScore=score;headerRow=r;}
  }
  var map=smartStaffColMap(rows[headerRow]);
  var start=headerRow;
  if(map.name!=null) start=headerRow+1;
  else map={name:0,title:1,jobTitle:2,employeeNo:3,birthDate:4,motherName:5,appointmentDate:6,startDate:7,phone:8,nationalId:9,status:10};
  var n=0;
  for(var r=start;r<rows.length;r++){
    var row=rows[r]||[];
    function cell(k){var i=map[k]; if(i==null)return ''; return String(row[i]==null?'':row[i]).trim();}
    var name=cell('name'); if(!name) continue;
    if(/^(ت|م|الاسم|اسم|رقم|#|name)$/i.test(name)) continue;
    if(/اسم المعلم|اسم المدرس|اسم الموظف|الاسم الكامل/.test(name) && !/\s/.test(name.replace(/اسم|المعلم|المدرس|الموظف|الكامل/g,''))) continue;
    var emp=(STAFF_STATE.staff||[]).find(function(e){return (typeof authNorm==='function'?authNorm(e.name):e.name)===(typeof authNorm==='function'?authNorm(name):name);});
    if(!emp){ emp={id:'emp_'+Date.now()+'_'+r,documents:[],increments:[],promotions:[]}; STAFF_STATE.staff.push(emp); }
    ['title','jobTitle','employeeNo','birthDate','motherName','appointmentDate','startDate','adminOrderDate','specialization','phone','nationalId','schoolStartDate','status'].forEach(function(k){ var v=cell(k); if(v) emp[k]=v; });
    emp.name=name;
    if(kind) emp.kind=kind;
    if(!emp.kind) emp.kind=(staffTableMode==='employees'?'employee':'teacher');
    if(!emp.jobTitle) emp.jobTitle=(emp.kind==='employee'?'موظف':'معلم');
    emp.updatedAt=Date.now();
    n++;
  }
  return n;
}
function exportStaffExcel(){
  if(typeof XLSX==='undefined'){alert('مكتبة الإكسل غير متاحة.');return;}
  const head=['ت','الاسم','اللقب','العنوان الوظيفي','الرقم الوظيفي','تاريخ الميلاد','اسم الأم','تاريخ التعيين','تاريخ المباشرة','تاريخ الأمر الإداري','الاختصاص','الهاتف','البطاقة الوطنية','المباشرة بالمدرسة','الحالة'];
  const keys=['name','title','jobTitle','employeeNo','birthDate','motherName','appointmentDate','startDate','adminOrderDate','specialization','phone','nationalId','schoolStartDate','status'];
  const wb=XLSX.utils.book_new();
  [['المعلمون','teacher'],['الموظفون','employee']].forEach(function(pair){
    const aoa=[head];
    (STAFF_STATE.staff||[]).filter(function(emp){return staffPersonKind(emp)===pair[1];}).forEach(function(emp,i){
      aoa.push([i+1].concat(keys.map(function(k){return emp[k]||'';})));
    });
    XLSX.utils.book_append_sheet(wb,XLSX.utils.aoa_to_sheet(aoa),pair[0]);
  });
  if(typeof xlsxWriteStyled==='function') xlsxWriteStyled(wb,'الملاك.xlsx'); else XLSX.writeFile(wb,'الملاك.xlsx');
}
function importStaffExcel(input){
  const file=input&&input.files&&input.files[0];
  if(input) input.value='';
  if(!file){return;}
  if(typeof XLSX==='undefined'){alert('مكتبة الإكسل غير متاحة.');return;}
  if(typeof authRole==='function'&&authRole()==='teacher'&&!teacherCanStaffEdit()){alert('غير مسموح باستيراد الملاك.');return;}
  const reader=new FileReader();
  reader.onload=function(ev){
    try{
      const wb=XLSX.read(ev.target.result,{type:'array'});
      let total=0, details=[];
      if(staffTableMode!=='employees') staffTableMode='info';
      wb.SheetNames.forEach(function(sn){
        const kind=/موظف|employee/i.test(sn)?'employee':(/معلم|مدرس|teacher/i.test(sn)?'teacher':'');
        const rows=XLSX.utils.sheet_to_json(wb.Sheets[sn],{header:1,raw:false,defval:''});
        const n=ingestStaffGrid(rows, kind||(staffTableMode==='employees'?'employee':'teacher'));
        if(n>0) details.push(sn+': '+n+' سجل');
        total+=n;
      });
      saveStaffState();
      if(staffTableMode==='employees')renderStaffEmployees(); else renderStaffInfo();
      if(total>0) alert('✅ تم استيراد '+total+' سجل بنجاح.');
      else alert('⚠️ لم يتم العثور على بيانات في الملف.');
    }catch(e){alert('❌ تعذر قراءة ملف الإكسل.');}
  };
  reader.readAsArrayBuffer(file);
}
function staffSplitPasteLine(line){
  var t=String(line||'').replace(/\ufeff/g,'');
  if(t.indexOf('\t')>=0) return t.split('\t');
  if(t.indexOf(';')>=0) return t.split(';');
  if(t.indexOf(',')>=0 && t.split(',').length>=2) return t.split(',');
  if(t.indexOf('،')>=0 && t.split('،').length>=2) return t.split('،');
  if(/\s{2,}/.test(t)) return t.split(/\s{2,}/);
  return [t];
}
