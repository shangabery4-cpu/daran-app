function bindSec(pane){
  function msg(t,c){ var m=pane.querySelector('[data-sec-msg]'); if(m){ m.textContent=t||''; m.className='nbset-msg'+(c?' '+c:''); } }
  function val(k){ var i=pane.querySelector('[data-sec="'+k+'"]'); return i?westDigits(i.value).trim():''; }
  function redraw(){ pane.innerHTML=secPaneHtml(); bindSec(pane); }
  pane.onclick=function(e){
    var b=e.target.closest('[data-sec]'); if(!b||b.tagName!=='BUTTON') return;
    var a=b.getAttribute('data-sec'), o=lockGet();
    if(a==='locknow'){ window.nbAppLockNow(); return; }
    if(a==='enable'){
      var len=Number((pane.querySelector('input[name="nbSecLen"]:checked')||{}).value||4), p1=val('n1'), p2=val('n2');
      if(!new RegExp('^[0-9]{'+len+'}$').test(p1)){ msg('الرمز يجب أن يكون '+ar(len)+' أرقام','err'); return; }
      if(p1!==p2){ msg('الرمزان غير متطابقين','err'); return; }
      lockSet({on:true,h:pinHash(p1),e:enc(p1),len:len,resume:false,mins:0,at:Date.now()}); redraw(); toast('🔐 تم تفعيل قفل التطبيق'); return;
    }
    if(a==='off'){
      if(!b.__sure){ b.__sure=true; b.textContent='اضغط مرة أخرى لتأكيد الإيقاف'; b.style.fontWeight='900'; setTimeout(function(){ if(b.isConnected){ b.__sure=false; b.textContent='إيقاف القفل الآن'; } },4000); return; }
      lockSet({on:false}); redraw(); toast('🔓 تم إيقاف قفل التطبيق'); return;
    }
    if(a==='change'){
      var n1=val('n1'), n2=val('n2');
      if(!/^([0-9]{4}|[0-9]{6})$/.test(n1)){ msg('الرمز الجديد يجب أن يكون ٤ أو ٦ أرقام','err'); return; }
      if(n1!==n2){ msg('الرمزان غير متطابقين — اكتب نفس الرمز في الخانتين','err'); return; }
      o.h=pinHash(n1); o.e=enc(n1); o.len=n1.length; o.on=true; lockSet(o); redraw(); toast('✅ تم تغيير رمز القفل'); var m2=pane.querySelector('.nbsec-t small'); if(m2) m2.textContent='تم حفظ الرمز الجديد ('+ar(n1.length)+' أرقام) — استخدمه عند فتح التطبيق.'; return;

    if(a==='mins'){ o.mins=Number(t.value)||0; lockSet(o); }  };

/* تسجيل البصمة الصوتية */
function Recorder(){
  var rec=null, chunks=[], stream=null, t0=0, mime='';
  return {
    start:function(){
      if(!navigator.mediaDevices||!navigator.mediaDevices.getUserMedia||!window.MediaRecorder) return Promise.reject(new Error('nomedia'));
      return navigator.mediaDevices.getUserMedia({audio:{echoCancellation:true,noiseSuppression:true}}).then(function(s){
        stream=s; chunks=[]; mime=['audio/webm;codecs=opus','audio/webm','audio/ogg;codecs=opus','audio/mp4'].filter(function(x){ try{ return MediaRecorder.isTypeSupported(x); }catch(e){ return false; } })[0]||'';
        rec=mime?new MediaRecorder(s,{mimeType:mime}):new MediaRecorder(s); rec.ondataavailable=function(e){ if(e.data&&e.data.size) chunks.push(e.data); }; rec.start(250); t0=now(); });
    },
    secs:function(){ return t0?(now()-t0)/1000:0; },
    stop:function(keep){ return new Promise(function(res){ if(!rec){ res(null); return; } var du=(now()-t0)/1000;
      rec.onstop=function(){ try{ stream.getTracks().forEach(function(t){ t.stop(); }); }catch(e){} if(!keep||du<0.7){ res(null); return; } var b=new Blob(chunks,{type:(rec.mimeType||mime||'audio/webm')}); res({blob:b,du:du}); };
      try{ rec.stop(); }catch(e){ res(null); } }); }
  };
}
