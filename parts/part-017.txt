/* ---------- التخزين: IndexedDB للنسخ الكاملة + فهرس صغير ---------- */
function idb(){
  return new Promise(function(res,rej){
    try{
      if(!window.indexedDB) return rej(new Error('IndexedDB غير متاح'));
      var r=indexedDB.open(DB_NAME,1);
      r.onupgradeneeded=function(){ var db=r.result; if(!db.objectStoreNames.contains(DB_STORE)) db.createObjectStore(DB_STORE,{keyPath:'id'}); };
      r.onsuccess=function(){ res(r.result); }; r.onerror=function(){ rej(r.error||new Error('IDB')); };
    }catch(e){ rej(e); }
  });
}
function idbDo(mode,fn){
  return idb().then(function(db){ return new Promise(function(res,rej){
    var tx=db.transaction(DB_STORE,mode), st=tx.objectStore(DB_STORE), out;
    var rq=fn(st); if(rq) rq.onsuccess=function(){ out=rq.result; };
    tx.oncomplete=function(){ res(out); }; tx.onerror=function(){ rej(tx.error); }; tx.onabort=function(){ rej(tx.error||new Error('abort')); };
  }); });
}

function ensureTargetSection(gradeLabel,secLabel,cache){
  var key=gradeLabel+'||'+secLabel; if(cache[key]) return cache[key];

  if(!Array.isArray(g.sections)) g.sections=[];

  if(!sec){
    sec={id:'s_'+Date.now().toString(36)+Math.random().toString(36).slice(2,6),label:secLabel,guide:'',students:[]};
    g.sections.push(sec);
    if(stg()==='professional'){

      var stObj=dep&&(dep.stages||[]).find(function(x){return x.name===p.slice(1).join(' — ');});

function applyNewYear(items,opts,newYear){
  APP.nbYearArchived={at:Date.now(),year:PLAN.year};
  var moves=[], grads=[];
  items.forEach(function(it){ if(it.action!=='stay') (it.action==='grad'?grads:moves).push(it); });
  // 1) سحب الطلاب المنتقلين والمتخرجين من شعبهم (نحتفظ بكائن الطالب كما هو: نفس المعرّف والصورة والبيانات)
  var taken={};
  moves.concat(grads).forEach(function(it){
    var g=findGrade(it.grade), sec=g&&(g.sections||[]).find(function(s){return s.id===it.secId;});
    if(!sec) return;
    var idx=(sec.students||[]).findIndex(function(s){return s.id===it.stId;});
    if(idx>=0){ taken[it.grade+'|'+it.secId+'|'+it.stId]=sec.students.splice(idx,1)[0]; }
  });
  // 2) إضافتهم إلى الصف التالي — نفس اسم الشعبة
  var cache={};
  moves.forEach(function(it){
    var obj=taken[it.grade+'|'+it.secId+'|'+it.stId]; if(!obj) return;
    obj.promotedFrom=it.grade; obj.lastYearResult=it.res.label; delete obj.repeater;
    ensureTargetSection(it.action.slice(3),it.secLabel,cache).students.push(obj);
  });
  items.forEach(function(it){
    if(it.action!=='stay') return;
    var g=findGrade(it.grade), sec=g&&(g.sections||[]).find(function(s){return s.id===it.secId;});
    var obj=sec&&(sec.students||[]).find(function(s){return s.id===it.stId;});
    if(obj){ obj.lastYearResult=it.res.label; if(it.res.key==='failed'||it.res.key==='complete') obj.repeater=true; }
  });
  // 3) ترتيب أبجدي
  if(opts.sort){
    stageGradeLabels().forEach(function(label){
      var g=findGrade(label); if(!g) return;
      (g.sections||[]).forEach(function(sec){ (sec.students||[]).sort(function(a,b){ return String(a.name||'').localeCompare(String(b.name||''),'ar'); }); });
    });
  }
  // 4) تفريغ درجات السنة (نفس الكائنات الحيّة حتى تعمل المزامنة بشكل صحيح)
  if(typeof GRADEBOOK!=='undefined' && GRADEBOOK){ clearObj(GRADEBOOK.records); GRADEBOOK.defaultDemoGradesV1=true; saveGradebook(GRADEBOOK); }
  if(opts.levels && typeof LEVELS_STATE!=='undefined' && LEVELS_STATE){ clearObj(LEVELS_STATE.records); if(typeof saveLevelsState==='function') saveLevelsState(); }
  if(opts.attendance && typeof ATTENDANCE_STATE!=='undefined' && ATTENDANCE_STATE){ ATTENDANCE_STATE.records.length=0; clearObj(ATTENDANCE_STATE.days); if(typeof saveAttendanceState==='function') saveAttendanceState(); }
  // 5) السنة الجديدة
  APP.school.academicYear=newYear;
  APP.nbYearArchived={at:Date.now(),year:PLAN.year};
  try{ if(typeof EXAMS_STATE!=='undefined'&&EXAMS_STATE&&EXAMS_STATE.seating){ EXAMS_STATE.seating.academicYear=newYear; if(typeof saveExamsState==='function') saveExamsState(); } }catch(e){}
  saveApp();
  return {moved:moves.length,grads:grads.length};
}

function db(){

function tx(store,mode,fn){
  return db().then(function(d){ return new Promise(function(res,rej){
    var t=d.transaction(store,mode), s=t.objectStore(store), out, rq=fn(s);
    if(rq) rq.onsuccess=function(){ out=rq.result; };
    t.oncomplete=function(){ res(out); }; t.onerror=function(){ rej(t.error); };
  }); });
}

function writeFolder(h,name,text,keep){
  return h.getFileHandle(name,{create:true}).then(function(fh){ return fh.createWritable(); })
    .then(function(w){ return w.write(new Blob([text],{type:'application/json'})).then(function(){ return w.close(); }); })
    .then(function(){ return pruneFolder(h,keep); }).then(function(){ return 'folder'; });
}
function pruneFolder(h,keep){
  if(!h.values) return Promise.resolve();
  var names=[];
  return (async function(){
    for await (var e of h.values()){ if(e.kind==='file' && e.name.indexOf(PREFIX)===0) names.push(e.name); }
    names.sort().reverse();
    for(var i=keep;i<names.length;i++){ try{ await h.removeEntry(names[i]); }catch(e){} }
  })();
}

var NAME_HEADS=/^(اسم الطالب|اسم التلميذ|اسم الطالبة|اسم الطالب\/ة|اسم التلميذ\/ة|الاسم|الطالب|التلميذ|اسم الطالب الرباعي|الاسم الثلاثي|الاسم الرباعي)$/;
function nameColumns(table){
  var grid=[], cols={};
  var rows=[].slice.call(table.tHead?table.tHead.rows:[]);
  if(!rows.length){ var fr=table.rows[0]; if(fr&&fr.querySelector('th')) rows=[fr]; }
  rows.forEach(function(tr,ri){
    grid[ri]=grid[ri]||[]; var c=0;
    [].forEach.call(tr.cells,function(cell){
      while(grid[ri][c]) c++;
      var cs=cell.colSpan||1, rs=cell.rowSpan||1;
      var t=(cell.textContent||'').replace(/\s+/g,' ').trim();
      if(NAME_HEADS.test(t)&&cs===1) cols[c]=1;
      for(var r=0;r<rs;r++) for(var k=0;k<cs;k++){ grid[ri+r]=grid[ri+r]||[]; grid[ri+r][c+k]=1; }
      c+=cs;
    });
  });
  return Object.keys(cols).map(Number);
}

/* اتجاه النص في ملفات PDF: علامات مثل «:» و«/» في نهاية النص العربي وبين الأرقام كانت تنقلب في PDF.
   نضيف علامة اتجاه غير مرئية (RLM) فيبقى الشكل كما على الشاشة تماماً. */
function fixBidi(doc){
  try{
    if(doc.__nbBidi) return; doc.__nbBidi=true;
    var win=doc.defaultView;
    var w=doc.createTreeWalker(doc.body,NodeFilter.SHOW_TEXT,null,false), n, list=[];
    while((n=w.nextNode())) list.push(n);
    var RE=/([:\/؟!،؛…]+|[\u0660-\u06690-9]+(?:[.,][\u0660-\u06690-9]+)?)/;
    list.forEach(function(t){
      var v=t.nodeValue; if(!v||!RE.test(v)) return;
      var el=t.parentElement; if(!el||el.closest('.qm,.qp-en,[dir="ltr"],script,style,sup,sub')) return;
      if(!/[\u0600-\u06FF]/.test(v)) return;              // نص عربي مع علامات/أرقام فقط
      try{ if(win.getComputedStyle(el).direction!=='rtl') return; }catch(e){ return; }
      // كل علامة أو رقم في عنصر مستقل: مولّد PDF يرسم كل عنصر في مكانه الصحيح كما في الشاشة
      var parts=v.split(RE); if(parts.length<2) return;
      var f=doc.createDocumentFragment();
      parts.forEach(function(x,i){ if(!x) return; if(i%2===1){ var sp=doc.createElement('span'); sp.textContent=x; f.appendChild(sp); } else f.appendChild(doc.createTextNode(x)); });
      t.parentNode.replaceChild(f,t);
    });
  }catch(e){}
}

var GLYPH_RE=/[\u00B1\u00B0\u00D7\u00F7\u0370-\u03FF\u2032\u2033\u2070-\u209F\u2100-\u214F\u2190-\u21FF\u2200-\u22FF\u2300-\u23FF\u25A0-\u25FF\u2713\u2717\u27F5-\u27FF\u2A00-\u2AFF]/;

  try{

      var f=doc.createDocumentFragment(), buf='';
      function flush(){ if(buf){ f.appendChild(doc.createTextNode(buf)); buf=''; } }
      Array.from(t.nodeValue).forEach(function(ch){
        if(!GLYPH_RE.test(ch)){ buf+=ch; return; }
        flush();
        var key=ch+'|'+fs+'|'+color+'|'+weight, src=GLYPH_CACHE[key], k=4;
        if(!src){
          var cv=doc.createElement('canvas'), x=cv.getContext('2d'); var font=weight+' '+fs+'px '+fam; x.font=font;
          var mw=Math.max(2,Math.ceil(x.measureText(ch).width)), h=Math.ceil(fs*1.3);

          src=GLYPH_CACHE[key]=cv.toDataURL('image/png')+'#w='+(mw+2)+'&h='+h;
        }
        var m=src.match(/#w=(\d+)&h=(\d+)$/), im=doc.createElement('img');

        im.style.cssText='display:inline-block;vertical-align:middle;width:'+m[1]+'px;height:'+m[2]+'px;margin:0;border:0';
        f.appendChild(im);
      });
      flush(); t.parentNode.replaceChild(f,t);
    });
  }catch(e){ try{console.warn('nb-glyph',e);}catch(_){} }
}
