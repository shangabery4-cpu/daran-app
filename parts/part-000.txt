function pack15(bytes){
  var n = bytes.length, out = [], acc = 0, bits = 0, chunk = [];
  for (var i = 0; i < n; i++) {
    acc = (acc << 8) | bytes[i]; bits += 8;
    if (bits >= 15) { bits -= 15; chunk.push(((acc >>> bits) & 0x7FFF) + 0x20); acc &= (1 << bits) - 1;
      if (chunk.length >= 8192) { out.push(String.fromCharCode.apply(null, chunk)); chunk = []; } }
  }
  if (bits > 0) chunk.push(((acc << (15 - bits)) & 0x7FFF) + 0x20);
  if (chunk.length) out.push(String.fromCharCode.apply(null, chunk));
  return out.join('');
}
function unpack15(str, byteLen){

  }
  return out;
}
