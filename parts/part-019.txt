function rasterizeFonts(d){
  var w=d.defaultView, K=4, cache={}, AR=/[؀-ۿݐ-ݿﭐ-﷿ﹰ-﻿]/;
  var walker=d.createTreeWalker(d.body,NodeFilter.SHOW_TEXT,null,false), n, list=[];
  while((n=walker.nextNode())) if(AR.test(n.nodeValue)) list.push(n);
  list.forEach(function(t){
    var p=t.parentElement; if(!p||p.closest('script,style,.nb-pframe,svg,textarea,select,option,button')) return;
    var cs=w.getComputedStyle(p), fam=String(cs.fontFamily||'').split(',')[0].replace(/["']/g,'').trim();
    if(!WEBFONT[fam.toLowerCase()]) return;
    if(cs.direction==='ltr'||(cs.writingMode&&cs.writingMode.indexOf('horizontal')<0)) return;
    var fs=parseFloat(cs.fontSize)||14, font=(cs.fontStyle==='italic'?'italic ':'')+(cs.fontWeight||'400')+' '+fs+'px "'+fam+'"', color=cs.color||'#000';
    var H=Math.ceil(fs*2), lh=fs*1.25, mv=-((H-lh)/2);
    var frag=d.createDocumentFragment();
    String(t.nodeValue).split(/(\s+)/).forEach(function(tok){
      if(!tok) return;
      if(/^\s+$/.test(tok)||!AR.test(tok)){ frag.appendChild(d.createTextNode(tok)); return; }
      var key=tok+'|'+font+'|'+color, c=cache[key];
      if(!c){
        var cv=document.createElement('canvas'), x=cv.getContext('2d'); x.font=font;
        var mw=Math.max(2,Math.ceil(x.measureText(tok).width))+2;
        cv.width=mw*K; cv.height=H*K; x=cv.getContext('2d'); x.scale(K,K); x.font=font; x.fillStyle=color; x.textBaseline='middle'; x.direction='rtl'; x.textAlign='right';
        x.fillText(tok,mw-1,H/2);
        c=cache[key]={src:cv.toDataURL('image/png'),w:mw};
      }
      var im=d.createElement('img'); im.src=c.src; im.alt=tok; im.setAttribute('data-nbtxt','1');
      im.style.cssText='display:inline-block;vertical-align:middle;width:'+c.w+'px;height:'+H+'px;margin:'+mv.toFixed(2)+'px 0;border:0;padding:0';
      frag.appendChild(im);
    });
    t.parentNode.replaceChild(frag,t);
  });
}

      if(m.section==null){ var sp2=splitGradeSection(r[m.grade]); if(sp2){ grade=sp2.grade; secLab=sp2.sec; } }
      if(!grade) grade=matchGrade(r[m.grade]);

function planDedup(plan){ /* لكل شعبة: كل طلاب الملف (بلا تكرار داخل الملف) + الجدد فقط + مطابقة الموجودين مسبقاً */
  if(plan.__dd) return; plan.__dd=true; plan.dup=0; plan.fileDup=0; plan.renamed=0;
  plan.order.forEach(function(k){
    var G=plan.groups[k];
    /* طالبان مختلفان يتشابهان في الاسم الثلاثي (مثل: كيان فلاح حسن عبيد / كيان فلاح حسن علوان): يُحفظ كلٌّ باسمه الرباعي حتى لا يُدمجا */

    function samePerson(a,b){ return N(a.st.__raw||a.st.name)===N(b.st.__raw||b.st.name)&&sameMother(a.st,b.st); }
    Object.keys(byN).forEach(function(n){ var L=byN[n]; if(L.length<2) return;
      var reps=[]; L.forEach(function(it){ if(!reps.some(function(r){ return samePerson(r,it); })) reps.push(it); });
      if(reps.length<2) return;
      L.forEach(function(it){ if(it.st.__raw&&N(it.st.__raw)!==N(it.st.name)){ it.short=it.st.name; it.st.name=it.st.__raw; it.renamed=true; plan.renamed++; } }); });
    var kept=[];
    G.all=G.items.filter(function(it){ if(kept.some(function(o){ return N(o.st.name)===N(it.st.name)&&sameMother(o.st,it.st); })){ plan.fileDup++; return false; } kept.push(it); return true; });
    G.all.sort(function(a,b){ return (a.seq!=null&&b.seq!=null)?(a.seq-b.seq||a.i-b.i):a.i-b.i; });
    var pool=G.sec?(G.sec.students||[]).filter(function(s){ return !s.demoDefault; }):[], used=new Set();
    G.existing=pool.length;
    G.all.forEach(function(it){
      var names=[N(it.st.name)]; if(it.short) names.push(N(it.short));
      var c=pool.filter(function(s){ return !used.has(s)&&names.indexOf(N(s.name))>=0&&sameMother(s,it.st); });
      c.sort(function(a,b){ return (N(b.name)===names[0])-(N(a.name)===names[0]); });
      it.old=c[0]||null; if(it.old) used.add(it.old);
    });
    G.items=G.all.filter(function(it){ if(it.old){ plan.dup++; return false; } return true; });
    G.removed=pool.length-used.size;
  });
}

function gridFromTable(tb){
  var out=[], occ=[];
  [].forEach.call(tb.rows,function(tr,ri){
    occ[ri]=occ[ri]||[]; var row=[], c=0;
    [].forEach.call(tr.cells,function(cell){
      while(occ[ri][c]!=null){ row[c]=''; c++; }
      var txt=(cell.innerText||cell.textContent||'').replace(/[ \s]+/g,' ').trim();
      var cs=cell.colSpan||1, rs=cell.rowSpan||1;
      for(var k=0;k<cs;k++){ row[c+k]=k===0?txt:''; for(var r=1;r<rs;r++){ occ[ri+r]=occ[ri+r]||[]; occ[ri+r][c+k]=1; } }
      c+=cs;
    });
    out.push(row);
  });
  return out;
}
function gridFromText(text){
  var lines=String(text||'').split(/\r?\n/).filter(function(l){ return l.trim(); });
  if(!lines.length) return [];
  var sep=lines.some(function(l){ return l.indexOf('\t')>=0; })?'\t':(lines.every(function(l){ return l.indexOf('|')>=0; })?'|':(lines.every(function(l){ return /;/.test(l); })?';':(lines.every(function(l){ return /,/.test(l); })?',':null)));
  return lines.map(function(l){ return sep?l.split(sep).map(function(c){ return c.trim(); }):l.split(/\s{2,}/).map(function(c){ return c.trim(); }); });
}
