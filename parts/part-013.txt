function wrapQuestionTextLines(text,maxChars){
  const result=[];String(text||'').split(/\r?\n/).forEach(function(paragraph){const words=paragraph.split(/\s+/).filter(Boolean);if(!words.length){result.push('');return;}let line='';words.forEach(function(word){if(!line)line=word;else if((line+' '+word).length<=maxChars)line+=' '+word;else{result.push(line);line=word;}});if(line)result.push(line);});return result.length?result:[''];
}
function buildQuestionPrintPages(data){
  const firstBudget=(isArtQuestionSubject(data.subject)||data.note)?230:240,continuationBudget=265,lineHeight=5.7,pages=[{questions:[],used:0,budget:firstBudget,first:true}];
  data.items.slice(0,data.count).forEach(function(rawItem,index){
    const item=normalizeQuestionItem(rawItem),combined=[item.text].concat(item.branches.map(function(branch,i){return QUESTION_BRANCH_LABELS[i]+'/ '+branch.text+' '+questionDegreeText(branch.degree);})).join('\n');
    let lines=wrapQuestionTextLines(combined,92),firstChunk=true;
    while(lines.length){
      let page=pages[pages.length-1],overhead=firstChunk?6:4,available=page.budget-page.used-overhead,maxLines=Math.floor(available/lineHeight);
      if(maxLines<1){page={questions:[],used:0,budget:continuationBudget,first:false};pages.push(page);continue;}
      const take=Math.min(lines.length,maxLines),chunkLines=lines.splice(0,take);
      const chunkHeight=overhead+take*lineHeight;
      page.questions.push({label:firstChunk?QUESTION_SHORT_LABELS[index]:'تكملة '+QUESTION_SHORT_LABELS[index],text:chunkLines.join('\n'),degree:firstChunk?item.degree:'',continuation:!firstChunk,height:chunkHeight});
      page.used+=chunkHeight;firstChunk=false;
      if(lines.length){pages.push({questions:[],used:0,budget:continuationBudget,first:false});}
    }
  });
  const endingHeight=40,last=pages[pages.length-1];
  if(last.used+endingHeight>last.budget){
    const next={questions:[],used:0,budget:continuationBudget,first:false};
    while(last.questions.length){const moved=last.questions.pop();last.used-=moved.height;next.questions.unshift(moved);next.used+=moved.height;if(next.used+endingHeight<=next.budget)break;}
    pages.push(next);
  }
  return pages;
}

function buildMonthlyDates(startDate,count,skipSaturday){
  const dates=[];let d=new Date((startDate||monthlyDefaultStartDate())+'T00:00:00');let safety=0;
  while(dates.length<count&&safety<90){
    const day=d.getDay();
    if(day!==5&&!(skipSaturday&&day===6))dates.push(localDateKey(d));
    d.setDate(d.getDate()+1);safety++;
  }
  return dates;
}

function monthlySubjectPair(val){
  if(!val) return {s1:'',s2:''};
  if(typeof val==='object'){
    return {s1:String(val.s1||val[0]||''),s2:String(val.s2||val[1]||'')};
  }
  const t=String(val);
  const i=t.indexOf('||');
  if(i>=0) return {s1:t.slice(0,i),s2:t.slice(i+2)};
  return {s1:t,s2:''};
}

function saveMonthlySubject(date,grade,slot,value){
  EXAMS_STATE.monthly.subjects=EXAMS_STATE.monthly.subjects||{};
  EXAMS_STATE.monthly.subjects[date]=EXAMS_STATE.monthly.subjects[date]||{};
  const cur=monthlySubjectPair(EXAMS_STATE.monthly.subjects[date][grade]);
  if(String(slot)==='2') cur.s2=value; else cur.s1=value;
  if(!cur.s1 && !cur.s2) delete EXAMS_STATE.monthly.subjects[date][grade];
  else if(!cur.s2) EXAMS_STATE.monthly.subjects[date][grade]=cur.s1;
  else if(!cur.s1) EXAMS_STATE.monthly.subjects[date][grade]=cur.s2;
  else EXAMS_STATE.monthly.subjects[date][grade]={s1:cur.s1,s2:cur.s2};
  saveExamsState();
}

function monthlyPrintableTable(){
  const source=document.querySelector('.monthly-exam-table');if(!source)return null;
  const table=source.cloneNode(true);
  const srcCells=source.querySelectorAll('td.monthly-subj-cell');
  table.querySelectorAll('td.monthly-subj-cell').forEach(function(td,idx){
    const srcTd=srcCells[idx];
    const names=[];
    if(srcTd){
      srcTd.querySelectorAll('select').forEach(function(sel){
        if(sel.value){
          const lab=sel.options[sel.selectedIndex]?sel.options[sel.selectedIndex].text:sel.value;
          if(lab) names.push(lab);
        }
      });
    }
    td.innerHTML=names.length?'<strong>'+names.map(function(n){return escapeHtml(n);}).join('<br>')+'</strong>':'';
  });
  const leftoverSrc=Array.from(source.querySelectorAll('td:not(.monthly-subj-cell) select'));
  table.querySelectorAll('td:not(.monthly-subj-cell) select').forEach(function(select,index){
    const original=leftoverSrc[index];
    const span=document.createElement('strong');
    span.textContent=original&&original.value?(original.options[original.selectedIndex]&&original.options[original.selectedIndex].text)||'':'';

  });

  });
  return table;
}

function printThemeSelectHtml(){
  return '<label class="print-theme-wrap"><span>قالب الألوان</span><select class="print-theme-select" onchange="setPrintTheme(this.value)">'+examColorThemeOptions(getPrintTheme())+'</select></label>';
}
function ensurePrintThemeBars(){
  try{
    if(typeof printThemeSelectHtml!=='function')return;
    if(typeof currentViewId!=='undefined'&&/view-(auth|mail|dashboard|teacher|student|privmail)$/.test(currentViewId||''))return;
    const views=document.querySelectorAll('.view');
    let host=null;
    views.forEach(function(v){if(v&&v.style&&v.style.display!=='none'&&v.offsetParent!==null)host=v;});
    if(!host)return;
    const skip=host.querySelector('.attendance-home-grid,.staff-home-grid,.care-home-grid,.acc-home-grid,.records-home-grid');
    const btns=Array.prototype.slice.call(host.querySelectorAll('button'));
    const printBtns=btns.filter(function(b){
      if(!b||!b.parentNode)return false;
      if(/module-card|grade-select-card|staff-module-card|care-module-card|attendance-module-card|monthly-exclude-mark/.test(b.className||''))return false;
      if(b.closest && (b.closest('table')||b.closest('.day-cell')||b.closest('.monthly-exam-table')))return false;
      const tx=(b.textContent||'').replace(/\s+/g,' ').trim();
      if(tx.length>32)return false;
      return /طباعة|🖨️|عمودي|أفقي/.test(tx);
    });
    printBtns.forEach(function(printBtn){
      const par=printBtn.parentNode;
      if(par.querySelector(':scope > .print-theme-wrap, :scope > span > .print-theme-wrap'))return;
      const box=document.createElement('span');
      box.style.cssText='display:inline-flex;align-items:center;margin-inline:6px;vertical-align:middle';
      box.innerHTML=printThemeSelectHtml();
      par.insertBefore(box, printBtn);
    });
  }catch(e){}

    const available=Object.keys(groups).filter(function(k){return groups[k].length;}).sort(function(a,b){return groups[b].length-groups[a].length;});

function recordFirstLetter(name){
  let s=String(name||'').trim();
  if(s.indexOf('ال')===0&&s.length>2)s=s.slice(2);
  const ch=s.charAt(0);
  if('اأإآٱء'.indexOf(ch)!==-1)return 'أ';
  if(ch==='ى'||ch==='ئ')return 'ي';
  if(ch==='ة')return 'ه';
  if(ch==='ؤ')return 'و';
  return RECORDS_LETTERS.indexOf(ch)!==-1?ch:'أ';
}

    if((maps.numCount[num]||0)>1||similarKeys[n]||(maps.nameCount[n]||0)>1&&similarKeys[n])conflictIds[r.id]=1;
    if((maps.numCount[num]||0)>1)conflictIds[r.id]=1;
    if(similarKeys[n])conflictIds[r.id]=1;
  });
  const withDocs=items.filter(recordsHasDoc).length;
  return {total:items.length,conflicts:Object.keys(conflictIds).length,duplicates:dupNames,withDocs:withDocs,letters:letterMap};
}
function recordsAnalysisItems(kind){
  const items=recordsItems(), maps=recordsDupMaps(), similar=recordsSimilarNameKeys();
  if(kind==='stored')return items.slice();
  if(kind==='docs')return items.filter(recordsHasDoc);
  if(kind==='dup')return items.filter(function(r){return (maps.nameCount[normalizeRecordName(r.name)]||0)>1;});
  if(kind==='conflict')return items.filter(function(r){
    const n=normalizeRecordName(r.name), num=String(r.recordNo||'').trim();
    return (num&&(maps.numCount[num]||0)>1)||!!similar[n];
  });
  return items.slice();
}
function recordsAnalysisTitle(kind){
  return kind==='conflict'?'القيود المتضاربة / المتشابهة':(kind==='dup'?'الأسماء المكررة':(kind==='docs'?'القيود المرفوع لها وثائق':'القيود المخزّنة'));
}

  }

function importRecordsExcel(input){
  const file=input&&input.files&&input.files[0];
  if(input) input.value='';
  if(!file)return;
  const reader=new FileReader();
  reader.onload=function(ev){
    try{
      if(typeof XLSX==='undefined'){alert('تعذر قراءة الإكسل في هذا المتصفح.');return;}
      const wb=XLSX.read(ev.target.result,{type:'array'});
      const sheet=wb.Sheets[wb.SheetNames[0]];
      const aoa=XLSX.utils.sheet_to_json(sheet,{header:1,raw:false,defval:''});
      if(!aoa||!aoa.length){alert('الملف فارغ.');return;}
      let start=0;
      const h=aoa[0].map(function(x){return String(x||'').trim();});
      if(/تسلسل|اسم|قيد/.test(h.join(' '))) start=1;
      let nameI=1,noI=2;
      h.forEach(function(c,i){
        if(/اسم/.test(c)) nameI=i;
        if(/قيد|رقم/.test(c)&&!/تسلسل/.test(c)) noI=i;
      });
      if(start===0){nameI=1;noI=2;}
      let added=0,updated=0;
      for(let r=start;r<aoa.length;r++){
        const row=aoa[r]||[];
        const name=String(row[nameI]!=null?row[nameI]:row[1]||'').trim();
        const recordNo=String(row[noI]!=null?row[noI]:row[2]||'').trim();
        if(!name&&!recordNo)continue;
        if(!name||!recordNo)continue;
        let rec=null;
        if(recordNo) rec=recordsItems().find(function(x){return String(x.recordNo||'')===recordNo;});
        if(!rec) rec=recordsItems().find(function(x){return normalizeRecordName(x.name)===normalizeRecordName(name);});
        if(rec){ rec.name=name; rec.recordNo=recordNo; updated++; }
        else { RECORDS_STATE.items.push({id:'rec_'+Date.now()+'_'+r,name:name,recordNo:recordNo,documentImage:'',grades:{},createdAt:new Date().toISOString()}); added++; }
      }
      rebuildRecordLetterSeq();saveRecordsState();renderRecordsList(RECORDS_FILTER_LETTER);
      alert('تم الاستيراد: '+added+' جديداً، '+updated+' محدّثاً.');
    }catch(e){alert('تعذرت قراءة ملف الإكسل.');}
  };
  reader.readAsArrayBuffer(file);
}

function parseRecordGradeValue(value){
  let raw=String(value==null?'':value);
  try{if(typeof toWesternDigits==='function')raw=toWesternDigits(raw);}catch(e){}
  raw=raw.replace(/[^\d.]/g,'').trim();
  if(!raw||raw==='.')return NaN;
  const n=parseFloat(raw);
  return Number.isFinite(n)?n:NaN;
}
function recordsAcademicYearOptions(selected){
  const years=[''];
  for(let y=1930;y<=2060;y++)years.push(y+' / '+(y+1));
  if(selected&&years.indexOf(selected)===-1)years.splice(1,0,selected);
  return years.map(function(year){
    const label=year?('\u200E'+year):'— فارغ —';
    return '<option value="'+escapeHtml(year)+'"'+(year===(selected||'')?' selected':'')+'>'+label+'</option>';
  }).join('');
}
function recordsGradeNumbers(rec){
  const nums=[];
  (recordGradeSubjects(rec)||[]).forEach(function(s){
    const n=parseRecordGradeValue(rec.grades&&rec.grades[s]);
    if(Number.isFinite(n))nums.push(n);
  });
  const sum=nums.reduce(function(a,b){return a+b;},0);
  const avg=nums.length?Math.round((sum/nums.length)*100)/100:0;
  return {sum:sum,avg:avg,count:nums.length};
}

function readNoteImage(){
  return new Promise(function(resolve){
    const inp=document.getElementById('noteImage');
    const file=inp&&inp.files&&inp.files[0];

    const r=new FileReader();

    r.onerror=function(){resolve(null);};
    r.readAsDataURL(file);
  });
}

function accNum(v){const n=parseFloat(String(v==null?'':v).replace(/[^\d.-]/g,''));return Number.isFinite(n)?n:0;}
function accMoney(v){return accNum(v).toLocaleString('ar-IQ');}
function accStudentPaid(row){
  if(row.payType==='installment'&&Array.isArray(row.installments)&&row.installments.length){
    return row.installments.reduce(function(s,it){return s+accNum(it.paid);},0);
  }
  return accNum(row.paid);
}
function accRemain(row){
  const paid=row&&row.payType==='installment'?accStudentPaid(row):accNum(row.paid);
  return accNum(row.due)-paid;
}
function accStatus(row){
  const paid=row&&row.payType==='installment'?accStudentPaid(row):accNum(row.paid);
  const r=accNum(row.due)-paid;
  if(accNum(row.due)<=0&&paid<=0)return 'فارغ';
  if(r<=0)return 'مسدد';
  if(row.payType==='installment'){
    const done=(row.installments||[]).filter(function(it){return accNum(it.paid)>=accNum(it.amount)&&accNum(it.amount)>0;}).length;
    if(done)return 'قسط '+done+'/'+(row.installments||[]).length;
    return 'أقساط';
  }
  if(paid>0)return 'جزئي';
  return row.payType==='cash'?'نقدي غير مسدد':'غير مسدد';
}
function accStatusClass(st){return st==='مسدد'?'acc-paid':(/غير مسدد/.test(st)?'acc-due':'acc-ok');}
function accPayLabel(row){return row.payType==='installment'?'أقساط':(row.payType==='cash'?'نقدي':'—');}
function ensureStudentInstallments(row){
  if(!row||row.payType!=='installment')return row;
  const n=Math.max(2,Math.min(12,parseInt(row.installmentCount,10)||4));
  row.installmentCount=n;
  const due=accNum(row.due);
  const each=n?Math.round((due/n)*100)/100:0;
  if(!Array.isArray(row.installments))row.installments=[];
  while(row.installments.length<n)row.installments.push({amount:each,paid:0,date:''});
  if(row.installments.length>n)row.installments=row.installments.slice(0,n);
  row.installments.forEach(function(it,i){if(!accNum(it.amount))it.amount=each;it.n=i+1;});
  row.paid=accStudentPaid(row);
  return row;
}

  let due=0,paid=0;
  (list||[]).forEach(function(r){due+=accNum(r.due);paid+=(r.payType==='installment'?accStudentPaid(r):accNum(r.paid));});

}

function saveAccountRow(kind,id){
  const name=(document.getElementById('accName').value||'').trim();
  if(!name){alert('أدخل الاسم.');return;}
  const row={name:name,due:accNum(document.getElementById('accDue').value),note:(document.getElementById('accNote').value||'').trim()};
  if(kind==='teachers'){
    row.job=(document.getElementById('accJob')||{}).value||'';
    row.paid=accNum(document.getElementById('accPaid').value);
  }else{
    row.grade=(document.getElementById('accGrade')||{}).value||'';
    row.section=(document.getElementById('accSection')||{}).value||'';
    row.payType=(document.getElementById('accPayType')||{}).value||'cash';
    if(row.payType==='installment'){
      row.installmentCount=parseInt((document.getElementById('accInstCount')||{}).value,10)||4;
      const each=row.installmentCount?Math.round((row.due/row.installmentCount)*100)/100:0;
      row.installments=[];
      document.querySelectorAll('.acc-inst-paid').forEach(function(inp,i){
        row.installments.push({n:i+1,amount:each,paid:accNum(inp.value),date:''});
      });
      while(row.installments.length<row.installmentCount)row.installments.push({n:row.installments.length+1,amount:each,paid:0,date:''});
      row.paid=accStudentPaid(row);
    }else{
      row.paid=accNum((document.getElementById('accPaid')||{}).value);
      row.installments=[];
    }
  }
  const list=kind==='teachers'?ACCOUNTS_STATE.teachers:ACCOUNTS_STATE.students;
  if(id){const cur=list.find(function(r){return r.id===id;});if(!cur)return;Object.assign(cur,row);}
  else{row.id='acc_'+Date.now();list.push(row);}
  saveAccountsState();closeAccountsForm();
  if(kind==='teachers')renderTeacherAccounts();else renderStudentAccounts();
}
