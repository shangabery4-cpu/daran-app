      r.onerror=function(){res();}; }catch(e){ res(); }
  });
}

function chemHtml(t){
  t=String(t||'').trim(); if(!t) return '';
  t=t.replace(/<=>|<->/g,'⇌').replace(/->|=>/g,'→').replace(/\(g\)/g,'<sub>(g)</sub>').replace(/\(l\)/g,'<sub>(l)</sub>').replace(/\(s\)/g,'<sub>(s)</sub>').replace(/\(aq\)/g,'<sub>(aq)</sub>');
  t=t.replace(/\^(\d*[+\-−])/g,function(m,c){ return '<sup>'+c.replace('-','−')+'</sup>'; });
  t=t.replace(/([A-Za-z\)\]])(\d+)/g,'$1<sub>$2</sub>');
  return '<span class="qm">'+t+'</span>&nbsp;';
}

/* ===== nb-v1.28: الدرجة النهائية + درجة القرار + الإكمال في كل السجلات =====
   1) الدرجة النهائية للصفوف غير الوزارية = (السعي السنوي + الامتحان النهائي) ÷ ٢ مقرّبة؛
      الصفوف الوزارية (السادس الابتدائي، الثالث المتوسط، السادس الإعدادي) = درجة الامتحان النهائي كما هي. المهني لا يتغير.

   3) الدرجة بعد القرار تُعتمد في النتيجة الرسمية والنسب والبطاقات والسجل الوسطي والفصلي والباركود.
   4) «الدرجة بعد الإكمال» إذا أُدخلت تُعتمد في النتيجة الرسمية (ناجح/مكمل/راسب) والنسب. */

function applyFinalRule(row,grade){
  if(!row||!ruleGrade(grade)) return;
  if(!(row._gradebookMonthlyV2||row._elem4)) return; // السجلات القديمة تحسب (السعي+النهائي)÷٢ أصلاً
  var ex=vs(row.result!=null&&row.result!==''?row.result:row.finalExam);
  if(!has(ex)){ row.finalGrade=''; return; }
  var cw=vs(row.coursework);
  row.finalGrade=(!isMinisterial(grade)&&has(cw))?Math.round((Number(cw)+Number(ex))/2):Number(ex);
}

var GUARD=0, CACHE=new Map(), clearSched=false;
function clearCache(){ CACHE.clear(); }
window.nbGbClearDecisionCache=clearCache;
function sig(rec){ return [rec.decisionPoints||0,rec.updatedAt||0,rec.term1BestTwo?1:0].join('|'); }
function fmt(n){ return Number(Number(n).toFixed(2)); }
function computeAllocations(grade,section,record){
  var allocations={};
  if(!section||!record) return allocations;
  var remaining=gradebookDecisionNumber(record.decisionPoints);
  if(remaining<=0) return allocations;
  var upper=usesSixthGradeLayout(grade), mark=resultPassMark(grade), candidates=[];
  GUARD++;
  try{
    (section.students||[]).forEach(function(st,index){
      var row=record.rows&&record.rows[st.id]; if(!row) return;
      if(upper) gbCompute56(row,record);
      var raw=gradebookNumeric(upper?row.finalGrade:row.finalExam);
      if(raw!=null&&raw<mark) candidates.push({id:st.id,raw:raw,need:fmt(mark-raw),index:index});
    });
  } finally { GUARD--; }
  candidates.sort(function(a,b){ return a.need-b.need||a.index-b.index; });
  candidates.forEach(function(item){
    if(remaining<=0) return;
    var applied=fmt(Math.min(remaining,item.need));
    if(applied>0){ allocations[item.id]={raw:item.raw,applied:applied,effective:fmt(item.raw+applied)}; remaining=fmt(remaining-applied); }
  });
  Object.keys(allocations).forEach(function(id){ allocations[id].remainingAfter=Math.max(0,remaining); });
  allocations.__remaining=remaining;
  return allocations;
}
function allocationsFor(record,grade,section){
  if(!record) return {};
  var c=CACHE.get(record);
  if(c&&c.sig===sig(record)) return c.allocs;
  grade=grade||recGrade(record); section=section||recSection(record);
  var allocs=computeAllocations(grade,section,record);
  CACHE.set(record,{sig:sig(record),allocs:allocs});
  if(!clearSched){ clearSched=true; setTimeout(function(){ clearSched=false; clearCache(); },0); }
  if(usesSixthGradeLayout(grade)){
    Object.keys(record.rows||{}).forEach(function(id){
      var row=record.rows[id]; if(!row||typeof row!=='object') return;
      var a=allocs[id];
      if(row.finalGradeRaw===undefined&&!a&&!row.decisionApplied) return;
      if(a){ row.finalGradeRaw=a.raw; row.finalGrade=a.effective; row.decisionApplied=a.applied; }
      else if(row.decisionApplied){ GUARD++; try{ gbCompute56(row,record); }catch(e){} finally{ GUARD--; } row.decisionApplied=0; row.finalGradeRaw=row.finalGrade; }
    });
  }
  return allocs;
}
window.gradebookDecisionAllocations=function(grade,section,record){ return allocationsFor(record,grade,section); };

function post(row,rec){
  if(!row||typeof row!=='object') return;
  var m=metaOf(row,rec&&rec.rows?rec:null); var R=m&&m.rec;
  var grade=recGrade(R);
  applyFinalRule(row,grade);
  if(!R||!usesSixthGradeLayout(grade)) return;

function grid(table){ /* خريطة الأعمدة المنطقية لكل خلية (مع الدمج) */
  var map=new Map(), occ=[];
  [].forEach.call(table.rows,function(tr,ri){
    occ[ri]=occ[ri]||[]; var c=0;
    [].forEach.call(tr.cells,function(cell){
      while(occ[ri][c]) c++;
      var cs=cell.colSpan||1, rs=cell.rowSpan||1;
      map.set(cell,{c:c,cs:cs,r:ri});
      for(var r=0;r<rs;r++){ occ[ri+r]=occ[ri+r]||[]; for(var k=0;k<cs;k++) occ[ri+r][c+k]=1; }
      c+=cs;
    });
  });
  var n=0; occ.forEach(function(row){ if(row&&row.length>n) n=row.length; });
  return {map:map,n:n};
}

function colWidth(deltaMm){
  var s=PV&&PV.sel; if(!s) return;
  var tables=sameTables(s.table), ws=colWidths(s.table), W=ws.reduce(function(a,b){ return a+b; },0); if(!W) return;
  snap();
  if(!PV.colInit) PV.colInit={key:clsKey(s.table)+'|'+ws.length,pcts:ws.map(function(w){ return w/W*100; })};
  var tmm=tableMm(s.table)||194, dp=deltaMm/tmm*100, pcts=ws.map(function(w){ return w/W*100; });
  var c=s.c, span=Math.max(1,s.cs), minP=Math.max(1.6,250/tmm);
  /* العمود المختار (أو مجموعة الأعمدة تحت عنوان مدمج) */
  var own=[], others=[]; for(var i=0;i<pcts.length;i++){ if(i>=c&&i<c+span) own.push(i); else others.push(i); }
  var ownSum=own.reduce(function(a,i){ return a+pcts[i]; },0);
  var want=Math.max(minP*own.length,ownSum+dp), give=want-ownSum;
  var room=others.reduce(function(a,i){ return a+Math.max(0,pcts[i]-minP); },0);
  if(give>room) { give=room; want=ownSum+give; }
  if(Math.abs(give)<0.01){ toast(dp>0?'لا توجد مساحة إضافية في الورقة':'وصل العمود إلى أصغر عرض'); PV.undo.pop(); return; }
  own.forEach(function(i){ pcts[i]=pcts[i]*(want/ownSum); });
  if(give>0){ others.forEach(function(i){ var r=Math.max(0,pcts[i]-minP); pcts[i]-=give*(r/room); }); }
