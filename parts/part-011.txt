function ensureDefaultDemoData(){  if(APP.defaultDemoDataV1) return;
  if(!Array.isArray(APP.grades)) APP.grades=[];
  GRADE_LABELS.forEach(function(label){
    let grade=APP.grades.find(function(g){return g.label===label;});
    if(!grade){
      grade={id:'g_'+gradeKey(label),label:label,sections:[]};
      APP.grades.push(grade);
    }
    if(!Array.isArray(grade.sections)) grade.sections=[];
    if(!grade.sections.length){
      grade.sections.push({id:'s_default_'+gradeKey(label),label:'أ',guide:demoGuideForSection(label,0),students:[]});
    }
    grade.sections.forEach(function(section,sectionIndex){
      if(!section.guide) section.guide=demoGuideForSection(label,sectionIndex);
      if(!Array.isArray(section.students)) section.students=[];
      const names=demoNamesForSection(label,sectionIndex);
      while(section.students.length<3){
        const i=section.students.length;
        section.students.push({
          id:'st_demo_'+gradeKey(label)+'_'+sectionIndex+'_'+i,
          name:names[i%3],
          demoDefault:true
        });
      }
    });
  });
  APP.defaultDemoDataV1=true;
  saveApp();
}

function currentUiSkin(){
  try{
    const s=localStorage.getItem(UI_SKIN_KEY);
    if(s&&UI_SKINS[s])return s;
    const old=localStorage.getItem(UI_THEME_KEY);
    if(old==='dark')return 'night-ember';
    if(old==='light')return 'day-ivory';
  }catch(e){}
  // Default for a new installation: the requested dark-night forest skin.
  // Explicitly saved user choices above are always preserved.
  return 'day-azure';
}

function applyUiSkin(skin){
  if(!UI_SKINS[skin])skin='night-pine';
  const mode=UI_SKINS[skin].mode;
  document.documentElement.setAttribute('data-ui-skin',skin);
  document.body.classList.toggle('theme-dark',mode==='dark');
  try{
    localStorage.setItem(UI_SKIN_KEY,skin);
    localStorage.setItem(UI_THEME_KEY,mode==='dark'?'dark':'light');
  }catch(e){}
  updateUiThemeControls();
}
function applyUiTheme(theme){
  const want=theme==='dark'?'dark':'light';
  let skin=currentUiSkin();
  if((UI_SKINS[skin]||{}).mode!==want){
    try{
      const last=localStorage.getItem(want==='dark'?'schoolResultsLastNightSkin_v1':'schoolResultsLastDaySkin_v1');
      if(last&&UI_SKINS[last]&&UI_SKINS[last].mode===want)skin=last;
      else skin=want==='dark'?'night-ember':'day-ivory';
    }catch(e){skin=want==='dark'?'night-ember':'day-ivory';}
  }
  applyUiSkin(skin);
}

function normalizeStudentSearchText(value){
  return String(value||'').toLowerCase().replace(/[ًٌٍَُِّْـ]/g,'').replace(/[أإآ]/g,'ا').replace(/ة/g,'ه').replace(/ى/g,'ي').replace(/\s+/g,' ').trim();

  const allowed={};visibleGradeGroups().forEach(function(group){group.labels.forEach(function(label){allowed[label]=1;});});

  (APP.grades||[]).forEach(function(grade){

function authClampView(id){
  const r=typeof authRole==='function'?authRole():'';
  if(r==='teacher'){
    if(id==='view-teacher'||id==='view-auth'||id==='view-attendance'||id==='view-exams') return id;
    if(id==='view-records'&&typeof teacherCanRecords==='function'&&teacherCanRecords()) return id;
    if((id==='view-home'||id==='view-students')&&typeof teacherCanSheets==='function'&&teacherCanSheets()) return id;
    if(id==='view-care'&&typeof teacherCanCare==='function'&&teacherCanCare()) return id;
    if(id==='view-parents'&&typeof teacherCanParents==='function'&&teacherCanParents()) return id;
    if(id==='view-timetable') return id;
    if(id==='view-staff') return id;
    return 'view-teacher';
  }
  if(r==='student' && id!=='view-student' && id!=='view-auth' && id!=='view-timetable') return 'view-student';
  return id;
}
function navigateTo(id, meta, replace){
  closeThemePicker();
  closeExitModal();
  closeToolbarPanels();
  const clamped=authClampView(id);
  if(clamped!==id){
    if(authRole()==='teacher'){renderTeacherHome();id='view-teacher';meta={page:'home'};}
    else if(authRole()==='student'){renderStudentHome();id='view-student';meta={page:'home'};}
  }
  showView(id);
  const entry = { id: id, meta: meta || {} };
  if(replace && appStack.length){
    appStack[appStack.length - 1] = entry;
    try{ history.replaceState({app:1, n:appStack.length}, ''); }catch(e){}
  } else {
    // If this exact page is already in the stack, this navigation is a "return" to it
    // (e.g. an on-screen «رجوع» button that re-opens the parent page). Cut the stack back
    // to that page instead of pushing a duplicate, so Back can never loop between two pages.
    var __k=appEntryKey(entry), __idx=-1;
    for(var __i=appStack.length-1;__i>=0;__i--){ if(appEntryKey(appStack[__i])===__k){ __idx=__i; break; } }
    if(__idx>=0){ appStack.length=__idx+1; appStack[__idx]=entry; }
    else appStack.push(entry);
    try{ history.pushState({app:1, n:appStack.length}, ''); }catch(e){}
  }
}

function appEntryKey(entry){
  if(!entry)return '';
  let meta='';
  try{meta=JSON.stringify(entry.meta||{});}catch(e){meta='';}
  return String(entry.id||'')+'|'+meta;
}
function appGoBack(){
  // A back action is always an in-app navigation step. It must never close the page.
  // Remove accidental duplicate history entries first; some menu layouts can
  // render the same view once before and once after navigation.
  if(Array.isArray(appStack)&&appStack.length>1){
    while(appStack.length>1&&appEntryKey(appStack[appStack.length-1])===appEntryKey(appStack[appStack.length-2])) appStack.pop();
  }
  if(document.getElementById('dashListOverlay')&&document.getElementById('dashListOverlay').classList.contains('open')){ closeDashNameList(); return true; }
  if(typeof isAccountsFormOpen==='function'&&isAccountsFormOpen()){ closeAccountsForm(); return true; }
  if(typeof isNotesFormOpen==='function'&&isNotesFormOpen()){ closeNotesForm(); return true; }
  if(typeof isRecordsFormOpen==='function'&&isRecordsFormOpen()){ closeRecordsForm(); return true; }
  if(typeof isRecordsDocOpen==='function'&&isRecordsDocOpen()){ closeRecordsDocViewer(); return true; }
  if(typeof isRecordsGradesOpen==='function'&&isRecordsGradesOpen()){ closeRecordsGradesForm(); return true; }
  if(typeof isCareFormOpen==='function'&&isCareFormOpen()){ closeCareForm(); return true; }
  if(typeof isStaffFormOpen==='function'&&isStaffFormOpen()){ closeStaffForm(); return true; }
  if(typeof isStudentRecordModalOpen==='function'&&isStudentRecordModalOpen()){ closeStudentRecordModal(); return true; }
  if(isStudentSearchOpen()){ studentSearchGoBack(); return true; }
  if(isGlobalNavOpen()){ closeGlobalNav(); return true; }
  const ov = document.getElementById('themeOverlay');
  if(ov && ov.classList.contains('open')){ closeThemePicker(); return true; }
  const ex = document.getElementById('exitModal');
  if(ex && ex.classList.contains('open')){ closeExitModal(); return true; }
  if(closeToolbarPanels()) return true;
  if(appStack.length > 1){
    appStack.pop();
    restoreEntry(appStack[appStack.length - 1]);
    return true;
  }
  // At the root, stay inside the dashboard instead of opening an exit dialog.
  if(currentViewId && currentViewId !== 'view-dashboard'){
    goDashboard();
    return true;
  }
  return true;
}
function requestAppBack(){
  // The global/menu back button is navigation only; exiting is available only through
  // an explicit exit action, never as a side effect of pressing Back.
  appGoBack();
}

function srPushCustomerRegistry(schoolName, principal, email){
  try{
    if(typeof window.srGetDeviceCode!=='function') return;
    const dev = window.srGetDeviceCode();
    const cleanDev = String(dev||'').replace(/[^0-9A-Z]/g,'');
    if(!cleanDev) return;
    const base = (typeof CLOUD_DEFAULT_URL!=='undefined' ? CLOUD_DEFAULT_URL : '').replace(/\/$/,'');
    if(!base) return;
    const url = base+'/_customers/'+encodeURIComponent(cleanDev)+'.json';
    const payload = {
      school: schoolName||'', principal: principal||'', email: email||'',
      device: dev||'', stage: (APP.school&&APP.school.stage)||'',
      registeredAt: Date.now()
    };
    fetch(url, {method:'PUT', headers:{'Content-Type':'application/json'}, body:JSON.stringify(payload)}).catch(function(){});
  }catch(e){}
}

function escapeHtml(s){
  return String(s||'').replace(/[&<>"']/g, c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));
}

function renderSubjectRows(key){
  const box = document.getElementById('subjList-'+key);
  if(!box) return;
  const list = getSubjects(key);
  box.innerHTML = '';
  list.forEach((name, idx)=>{
    const row = document.createElement('div');
    row.className = 'subj-row';
    row.innerHTML =
      '<button type="button" title="أعلى" onclick="moveSubjectItem(\''+key+'\','+idx+',-1)">↑</button>' +
      '<button type="button" title="أسفل" onclick="moveSubjectItem(\''+key+'\','+idx+',1)">↓</button>' +
      '<input type="text" value="'+escapeHtml(name)+'" oninput="updateSubjectItem(\''+key+'\','+idx+',this.value)">' +
      '<button type="button" class="del" onclick="removeSubjectItem(\''+key+'\','+idx+')">حذف</button>';
    box.appendChild(row);
  });
}
function updateSubjectItem(key, idx, val){
  ensureSubjects();
  if(!APP.subjects[key] || !APP.subjects[key][idx] && APP.subjects[key][idx] !== '') return;
  APP.subjects[key][idx] = val;
  saveApp();
}
function addSubjectItem(key){
  const input = document.getElementById('subjNew-'+key);
  const val = input ? input.value.trim() : '';
  if(!val) return;
  ensureSubjects();
  APP.subjects[key].push(val);
  saveApp();
  if(input) input.value = '';
  renderSubjectRows(key);
}
function removeSubjectItem(key, idx){
  ensureSubjects();
  if(APP.subjects[key].length <= 1){ alert('يجب أن تبقى مادة واحدة على الأقل.'); return; }
  if(!confirm('حذف هذه المادة من قائمة المرحلة؟')) return;
  APP.subjects[key].splice(idx, 1);
  saveApp();
  renderSubjectRows(key);
}
function moveSubjectItem(key, idx, dir){
  ensureSubjects();
  const list = APP.subjects[key];
  const j = idx + dir;
  if(j < 0 || j >= list.length) return;
  const tmp = list[idx]; list[idx] = list[j]; list[j] = tmp;
  saveApp();
  renderSubjectRows(key);
}
function resetSubjectSet(key){
  if(!confirm('إعادة المواد الافتراضية لهذه القائمة؟ سيتم استبدال قائمتك الحالية.')) return;
  ensureSubjects();
  APP.subjects[key] = DEFAULT_SUBJECT_SETS[key].items.slice();
  saveApp();
  renderSubjectRows(key);
}

function toggleGrade(label, on){
  const area = document.getElementById('sections-'+gradeKey(label));
  if(on){
    area.style.display = 'block';
    if(!APP.grades.find(g=>g.label===label)) APP.grades.push({id:'g_'+label, label, sections:[]});

}

function addSection(label){
  const input = document.getElementById('newSection-'+gradeKey(label));
  const val = input.value.trim();
  if(!val) return;
  let g = APP.grades.find(g=>g.label===label);
  if(!g){ g = {id:'g_'+label, label, sections:[]}; APP.grades.push(g); }
  if((g.sections||[]).some(function(sec){return sec.label===val;})){
    alert('هذه الشعبة مضافة مسبقًا لهذا الصف.');
    return;
  }
  const newSectionId='s_'+Date.now()+Math.random().toString(36).slice(2,6);
  const sectionIndex=g.sections.length;
  const defaultNames=demoNamesForSection(label,sectionIndex);
  const newSection={
    id:newSectionId,
    label:val,
    guide:demoGuideForSection(label,sectionIndex),
    students:[
      {id:'st_'+Date.now()+'_1_'+Math.random().toString(36).slice(2,5),name:defaultNames[0],demoDefault:true},
      {id:'st_'+Date.now()+'_2_'+Math.random().toString(36).slice(2,5),name:defaultNames[1],demoDefault:true},
      {id:'st_'+Date.now()+'_3_'+Math.random().toString(36).slice(2,5),name:defaultNames[2],demoDefault:true}
    ]
  };
  g.sections.push(newSection);
  if(typeof seedDemoGradesForSection==='function'){
    seedDemoGradesForSection(label,newSection);
    saveGradebook(GRADEBOOK);
  }
  input.value = '';
  saveApp();
  renderChips(label);
  renderManagedSectionCards();
}

function xlsxHeaderStyle(){

  return {font:{name:'Tahoma',sz:11,color:{rgb:'21151A'}},fill:{patternType:'solid',fgColor:{rgb:even?'FFF3D6':'FFFFFF'}},alignment:{horizontal:'center',vertical:'center',wrapText:true},border:{top:{style:'thin',color:{rgb:'B0A080'}},bottom:{style:'thin',color:{rgb:'B0A080'}},left:{style:'thin',color:{rgb:'B0A080'}},right:{style:'thin',color:{rgb:'B0A080'}}}};

function styleExcelSheet(ws){
  if(!ws||!ws['!ref'])return ws;
  const range=XLSX.utils.decode_range(ws['!ref']);
  const cols=[];
  for(let C=range.s.c;C<=range.e.c;C++){
    let max=10;
    for(let R=range.s.r;R<=range.e.r;R++){
      const addr=XLSX.utils.encode_cell({r:R,c:C});
      const cell=ws[addr];
      if(!cell)continue;
      cell.s=(R===range.s.r)?xlsxHeaderStyle():xlsxCellStyle((R-range.s.r)%2===0);
      const len=String(cell.v==null?'':cell.v).length;
      if(len+2>max) max=Math.min(len+2,36);
    }
    cols.push({wch:max});
  }
  ws['!cols']=cols;
  ws['!rows']=[{hpt:24}];
  return ws;
}
function xlsxWriteStyled(wb, filename){
  if(!wb||typeof XLSX==='undefined')return;
  /* أسماء الصفوف المعدّلة تظهر في ملفات الإكسل أيضاً */
  try{ if(typeof window.nbGradeNameText==='function'){ (wb.SheetNames||[]).forEach(function(n){ var ws=wb.Sheets[n]; Object.keys(ws).forEach(function(k){ if(k.charAt(0)==='!') return; var c=ws[k]; if(c&&c.t==='s'&&typeof c.v==='string'){ var v2=window.nbGradeNameText(c.v); if(v2!==c.v){ c.v=v2; if(c.w!=null) c.w=v2; if(c.h!=null) delete c.h; } } }); }); filename=window.nbGradeNameText(String(filename)); } }catch(e){}
  (wb.SheetNames||[]).forEach(function(n){ try{styleExcelSheet(wb.Sheets[n]);}catch(e){} try{ wb.Sheets[n]['!views']=[{RTL:true}]; }catch(e){} });
  try{ wb.Workbook=wb.Workbook||{}; wb.Workbook.Views=[{RTL:true}]; }catch(e){} /* nb-v1.48.2: كل ملفات الإكسل من اليمين إلى اليسار */
  XLSX.writeFile(wb, filename);
}

function isPrimaryTenMarkGrade(grade){
  return grade==='الأول'||grade==='الثاني'||grade==='الثالث'||grade==='الرابع';
}
function resultFullMark(grade){ return isPrimaryTenMarkGrade(grade)?10:100; }
function resultPassMark(grade){ return isPrimaryTenMarkGrade(grade)?5:50; }
function resultRules(){
  const c=Math.max(1,parseInt(toWesternDigits((APP.school&&APP.school.completeMax)!=null?APP.school.completeMax:2),10)||2);
  const f=Math.max(1,parseInt(toWesternDigits((APP.school&&APP.school.failMin)!=null?APP.school.failMin:3),10)||3);
  return {completeMax:c,failMin:f};
}
function studentFinalScoreList(grade,sectionId,studentId){
  const subjects=typeof gbSubjects==='function'?gbSubjects(grade):[];
  const list=[];
  subjects.forEach(function(subject){
    const record=typeof gbRecord==='function'?gbRecord(grade,sectionId,subject,false):null;
    const row=record&&record.rows?record.rows[studentId]:null;
    if(!row)return;
    let value='';
    if(typeof usesSixthGradeLayout==='function'&&usesSixthGradeLayout(grade)){
      if(typeof gbCompute56==='function')gbCompute56(row,record);
      value=row.finalGrade;
    }else value=row.finalExam;
    const n=Number(toWesternDigits(value));
    if(value!==''&&value!=null&&Number.isFinite(n))list.push(n);
  });
  return list;
}
function classifyStudentResult(scores,grade){
  if(!scores||!scores.length)return {key:'pending',label:'—'};
  const mark=resultPassMark(grade);
  const failN=scores.filter(function(n){return Number(n)<mark;}).length;
  const rules=resultRules();
  if(failN===0)return {key:'passed',label:'ناجح'};
  if(failN>=rules.failMin)return {key:'failed',label:'راسب'};
  if(failN<=rules.completeMax)return {key:'complete',label:'مكمل'};
  return {key:'failed',label:'راسب'};
}
function studentOfficialResult(grade,sectionId,studentId){
  return classifyStudentResult(studentFinalScoreList(grade,sectionId,studentId),grade);
}

function exemptionNumber(value,fallback){
  const n=Number(toWesternDigits(String(value==null?'':value)).replace(/[^0-9.\-]/g,''));
  return Number.isFinite(n)?n:fallback;
}
function exemptionClamp(value,fallback){return Math.max(0,Math.min(100,exemptionNumber(value,fallback)));}
let EXEMPTIONS_STATE=loadExemptionsState();
if(!Number.isFinite(Number(EXEMPTIONS_STATE.individualThreshold)))EXEMPTIONS_STATE.individualThreshold=90;
if(!Number.isFinite(Number(EXEMPTIONS_STATE.collectiveThreshold)))EXEMPTIONS_STATE.collectiveThreshold=75;

  try{if(typeof wifiOnSave==='function')wifiOnSave('exemptions');}catch(e){}

function exemptionThreshold(kind){
  return exemptionClamp(kind==='collective'?EXEMPTIONS_STATE.collectiveThreshold:EXEMPTIONS_STATE.individualThreshold,kind==='collective'?75:90);
}
function exemptionScoreNumber(value){
  if(value==null||String(value).trim()==='')return null;
  const n=Number(toWesternDigits(String(value)).replace(/[^0-9.\-]/g,''));
  return Number.isFinite(n)?n:null;
}
function exemptionSubjectScore(grade,sectionId,studentId,subject){
  const record=typeof gbRecord==='function'?gbRecord(grade,sectionId,subject,false):null;
  const row=record&&record.rows?record.rows[studentId]:null;
  if(!row)return null;
  let value='';
  if(usesSixthGradeLayout(grade)){
    if(typeof gbCompute56==='function')gbCompute56(row,record);
    value=row.finalGrade;
  }else value=row.finalExam;
  const n=exemptionScoreNumber(value);
  if(n==null)return null;
  const full=typeof resultFullMark==='function'?resultFullMark(grade):100;
  return {score:n,full:full,percent:full?((n/full)*100):0,display:showGrade(n)+' / '+showGrade(full)};
}
function exemptionQualified(score,kind,grade){
  return !!score && score.percent >= exemptionThreshold(kind);
}
function isStudentCollectivelyExempt(grade,sectionId,studentId){
  const subjects=(typeof gbSubjects==='function'?gbSubjects(grade):[]).filter(function(s){return String(s||'').trim();});
  if(!subjects.length)return false;
  return subjects.every(function(subject){return exemptionQualified(exemptionSubjectScore(grade,sectionId,studentId,subject),'collective',grade);});
}
function studentHasAnyIndividualExemption(grade,sectionId,studentId){
  const subjects=(typeof gbSubjects==='function'?gbSubjects(grade):[]).filter(function(s){return String(s||'').trim();});
  return subjects.some(function(subject){return exemptionQualified(exemptionSubjectScore(grade,sectionId,studentId,subject),'individual',grade);});
}
function studentHasSubjectIndividualExemption(grade,sectionId,studentId,subject){
  if(subject==='القائمة الموحدة')return studentHasAnyIndividualExemption(grade,sectionId,studentId);
  return exemptionQualified(exemptionSubjectScore(grade,sectionId,studentId,subject),'individual',grade);
}
function isExemptionFinalRound(type){
  const t=String(type||'');

}
function shouldExcludeStudentFromExamList(grade,sectionId,studentId,subject,type){
  if(!isExemptionFinalRound(type))return false;
  return isStudentCollectivelyExempt(grade,sectionId,studentId)||studentHasSubjectIndividualExemption(grade,sectionId,studentId,subject);
}
function exemptionReport(){
  const individual=[],collective=[];
  const grades=typeof gradebookGradeLabels==='function'?gradebookGradeLabels():[];
  grades.forEach(function(grade){
    gbSectionsForGrade(grade).forEach(function(section){
      (section.students||[]).filter(function(st){return String(st.name||'').trim();}).forEach(function(student){
        const subjects=(typeof gbSubjects==='function'?gbSubjects(grade):[]).filter(function(s){return String(s||'').trim();});
        subjects.forEach(function(subject){
          const score=exemptionSubjectScore(grade,section.id,student.id,subject);
          if(exemptionQualified(score,'individual',grade))individual.push({grade:grade,section:section,student:student,subject:subject,score:score});
        });
        if(subjects.length&&isStudentCollectivelyExempt(grade,section.id,student.id)){
          subjects.forEach(function(subject){
            const score=exemptionSubjectScore(grade,section.id,student.id,subject);
            if(score)collective.push({grade:grade,section:section,student:student,subject:subject,score:score});
          });
        }
      });
    });
  });
  return {individual:individual,collective:collective};
}

function calculateStudentAge(birthDate){
  if(typeof ymdDiff==='function'){
    const d=ymdDiff(birthDate);
    if(!d||d.empty)return '';
    if(typeof formatYmd==='function')return formatYmd(d);
    return d.years+' سنة، '+d.months+' شهر، '+d.days+' يوم';
  }
  if(!birthDate)return '';
  const born=new Date(birthDate+'T00:00:00');
  if(isNaN(born.getTime()))return '';
  const today=new Date();let age=today.getFullYear()-born.getFullYear();
  const month=today.getMonth()-born.getMonth();
  if(month<0||(month===0&&today.getDate()<born.getDate()))age--;
  return age>=0?age:'';
}

function saveBulkStudentTransfer(passedOnly){
  const source=getCurrentSection();if(!source)return;
  const gradeLabel=document.getElementById('transferGrade').value;
  const sectionId=document.getElementById('transferSection').value;
  if(!sectionId){alert('لا توجد شعبة في الصف المحدد. أضف شعبة أولاً.');return;}
  if(gradeLabel===currentSectionCtx.gradeLabel&&sectionId===currentSectionCtx.sectionId){alert('اختر شعبة مختلفة.');return;}
  const targetGrade=APP.grades.find(function(g){return g.label===gradeLabel;});
  const target=targetGrade&&targetGrade.sections.find(function(sec){return sec.id===sectionId;});if(!target)return;
  const moving=[];
  source.students.slice().forEach(function(student){
    if(passedOnly){
      const res=studentOfficialResult(currentSectionCtx.gradeLabel,currentSectionCtx.sectionId,student.id);
      if(res.key!=='passed') return;
    }
    moving.push(student);
  });
  if(!moving.length){alert(passedOnly?'لا يوجد ناجحون للترحيل.':'لا يوجد طلاب للترحيل.');return;}
  if(!confirm('ترحيل '+moving.length+' طالب إلى الصف '+gradeLabel+' شعبة '+target.label+'؟'))return;
  moving.forEach(function(student){
    student.transferHistory=student.transferHistory||[];
    student.transferHistory.push({fromGrade:currentSectionCtx.gradeLabel,fromSection:source.label,toGrade:gradeLabel,toSection:target.label,date:new Date().toISOString(),bulk:true,passedOnly:!!passedOnly});
    source.students=source.students.filter(function(s){return s.id!==student.id;});
    student.status='منقول';
    target.students.push(student);
  });
  saveApp();closeStudentRecordModal();
  if(currentViewId==='view-students') renderStudentsView();
  else renderHomeView();
}

function normalizeImportedStudentName(value){
  const text=String(value||'').replace(/\s+/g,' ').trim();
  if(!text)return '';
  const tokens=text.split(' ');
  const units=[];
  const twoWordCompounds=new Set([
    'نور الهدى','زين العابدين','سيف الدين','صلاح الدين','ضياء الدين','علاء الدين',
    'تاج الدين','شمس الدين','فخر الدين','نجم الدين','عز الدين','محب الدين',
    'عبد الله','عبد الرحمن','عبد الرحيم','عبد الكريم','عبد العزيز','عبد الوهاب',
    'عبد الرزاق','عبد القادر','عبد الجبار','عبد الستار','عبد السلام','عبد الواحد',
    'عبد اللطيف','عبد الأمير','عبد الزهرة','عبد الحسين','عبد النبي','عبد الرسول',
    'أبو الحسن','أبو الفضل','أبو القاسم','ابن سينا',
    'محمد رضا','محمد باقر','محمد جواد','محمد مهدي','علي أكبر','علي رضا',
    'فاطمة الزهراء','أم البنين','نور الزهراء','قمر الزمان'
  ]);
  for(let i=0;i<tokens.length&&units.length<3;){
    const pair=i+1<tokens.length?(tokens[i]+' '+tokens[i+1]):'';
    const dynamicCompound=(tokens[i]==='عبد'||tokens[i]==='أبو'||tokens[i]==='أم')&&i+1<tokens.length;
    if(pair&&(twoWordCompounds.has(pair)||dynamicCompound)){
      units.push(pair);i+=2;
    }else{
      units.push(tokens[i]);i+=1;
    }
  }
  return units.join(' ');
}

function smartStudentColMap(headers){
  const map={};
  function normH(c){return String(c||'').replace(/\s+/g,' ').replace(/[\u200f\u200e]/g,'').trim();}
  (headers||[]).forEach(function(c,i){
    const t=normH(c);
    if(!t)return;
    if(/قيد/.test(t)&&map.registrationNo==null) map.registrationNo=i;
    else if(/لقب/.test(t)) map.title=i;
    else if(/هاتف|جوال|موبايل|ولي\s*الأمر|ولي\s*الامر|رقم\s*ولي/.test(t)) map.guardianPhone=i;
    else if(/اسم\s*الأم|اسم\s*الام|اسم الأم|الأم$|الام$/.test(t) && !/هاتف|ولي\s*الأ/.test(t)) map.motherName=i;
    else if(/مواليد|ميلاد|تولد/.test(t) && !/محل|مسقط|مكان/.test(t)) map.birthDate=i;
    else if(/وثيق/.test(t)&&map.documentNo==null) map.documentNo=i;
    else if(/بطاقة|وطني|هوية/.test(t)) map.nationalId=i;
    else if(/مسقط|محل الولادة|مكان الولادة|مسقط الرأس/.test(t)) map.birthPlace=i;
    else if(/رعاية/.test(t)) map.careCoverage=i;
    else if(/حالة/.test(t)) map.status=i;
    else if(/عمر/.test(t)) map.age=i;
    else if(/^(ت|#|تسلسل)$/.test(t)) map.seq=i;
    else if(/اسم\s*(الطالب|التلميذ|التلمي)/.test(t) && map.name==null) map.name=i;
    else if(/^(الاسم)$/.test(t) && map.name==null) map.name=i;
  });
  if(map.name==null){
    (headers||[]).forEach(function(c,i){
      const t=normH(c);
      if(/تلميذ|طالب/.test(t) && !/أم|ام|هاتف/.test(t) && map.name==null) map.name=i;
    });
  }
  return map;
}
function looksLikePhone(v){return /^(07|7|9647|\+9647)\d{8,10}$/.test(String(v||'').replace(/\s+/g,''));}
function looksLikeId(v){const d=String(v||'').replace(/\D/g,'');return d.length>=10&&d.length<=16&&!looksLikePhone(v);}
function looksLikeDate(v){return /\d{1,4}[\/\-.\-]\d{1,2}[\/\-.\-]\d{1,4}/.test(String(v||''))||/^\d{5}$/.test(String(v||'').trim());}
function normalizeImportDate(v){
  v=String(v||'').trim();
  if(!v)return '';
  if(/^[0-9]{5,6}$/.test(v) && typeof XLSX!=='undefined'&&XLSX.SSF){
    try{const d=XLSX.SSF.parse_date_code(Number(v)); if(d) return String(d.y)+'-'+String(d.m).padStart(2,'0')+'-'+String(d.d).padStart(2,'0');}catch(e){}
  }
  const m=v.match(/^(\d{1,2})[\/\-.](\d{1,2})[\/\-.](\d{2,4})$/);
  if(m){
    let a=m[1],b=m[2],y=m[3];
    if(y.length===2) y=(Number(y)>50?'19':'20')+y;
    if(Number(a)>12) return y+'-'+String(b).padStart(2,'0')+'-'+String(a).padStart(2,'0');
    return y+'-'+String(a).padStart(2,'0')+'-'+String(b).padStart(2,'0');
  }
  return v;
}
function studentFromMappedRow(row, map){
  function cell(key, fb){ const i=map[key]; if(i==null) return fb||''; return String(row[i]==null?'':row[i]).trim(); }
  let name=normalizeImportedStudentName(cell('name'));
  if(!name) return null;
  let title=cell('title'), mother=cell('motherName'), birth=cell('birthDate'), phone=cell('guardianPhone');
  let place=cell('birthPlace'), nid=cell('nationalId'), status=cell('status')||'مستمر', care=cell('careCoverage');
  if(looksLikePhone(mother) && !looksLikePhone(phone)){ const t=phone; phone=mother; mother=t; }
  if(looksLikePhone(place) && !phone){ phone=place; place=''; }
  if(looksLikeId(birth) && !looksLikeId(nid)){ const t=nid; nid=birth; birth=t; }
  if(looksLikeDate(place) && !looksLikeDate(birth)){ const t=birth; birth=place; place=t; }
  if(looksLikeId(mother) && !nid){ nid=mother; mother=''; }
  birth=normalizeImportDate(birth);
  nid=String(nid||'').replace(/\.0$/,'');
  phone=String(phone||'').replace(/\.0$/,'');
  return {
    id:'st_'+Date.now()+Math.random().toString(36).slice(2,6),
    name:name, title:title, motherName:mother, birthDate:birth, guardianPhone:phone,
    status:status||'مستمر', birthPlace:place, nationalId:nid,
    careCoverage:/مشمول/.test(care)&&!/غير/.test(care)?'مشمول':'غير مشمول',
    documentNo:(function(){ var v=String(cell('documentNo')||'').replace(/\.0$/,'').trim(); try{ v=toWesternDigits(v); }catch(e){} return v; })(),
    registrationNo:(function(){ var v=String(cell('registrationNo')||'').replace(/\.0$/,'').trim(); try{ v=toWesternDigits(v); }catch(e){} return v.replace(/\s+/g,''); })()
  };
}
function ingestStudentGrid(rows, sec){
  if(!sec||!rows||!rows.length) return 0;
  let start=0, map={}, headerish=false;
  for(let h=0;h<Math.min(3,rows.length);h++){
    const cand=smartStudentColMap(rows[h]);
    if(cand.name!=null && Object.keys(cand).length>=2){ map=cand; start=h+1; headerish=true; break; }
  }
  if(!headerish) map={name:0,title:1,motherName:2,birthDate:3,guardianPhone:4,status:5,birthPlace:6,nationalId:7,careCoverage:8};
  let n=0;
  for(let r=start;r<rows.length;r++){
    const st=studentFromMappedRow(rows[r]||[], map);
    if(!st) continue;
    sec.students.push(st); n++;
  }
  return n;
}

function importStudentsFromWorkbook(wb, allSchool){
  if(!wb||!wb.SheetNames||!wb.SheetNames.length) return 0;
  let added=0, used=0;
  if(allSchool || wb.SheetNames.length>1){
    wb.SheetNames.forEach(function(sn){
      const found=typeof backupFindSection==='function'?backupFindSection(sn):null;
      if(!found||!found.sec) return;
      const rows=XLSX.utils.sheet_to_json(wb.Sheets[sn],{header:1,raw:false,defval:''});
      added+=ingestStudentGrid(rows, found.sec);
      used++;
    });
  }
  if(!used){
    const sec=getCurrentSection();
    if(!sec) return 0;
    const rows=XLSX.utils.sheet_to_json(wb.Sheets[wb.SheetNames[0]],{header:1,raw:false,defval:''});
    added+=ingestStudentGrid(rows, sec);
    used=1;
  }
  return added;
}

      const wb = XLSX.read(new Uint8Array(ev.target.result), { type:'array' });

    }
  };
  reader.readAsArrayBuffer(file);

function dashboardGradeStatistics(grade){

  sections.forEach(function(section){

      const latest=studentFinalScoreList(grade,section.id,student.id);

}

        ranked.push({id:student.id,name:student.name,grade:grade,section:section.label,avg:avg,count:scores.length});
      });
    });
  });
  ranked.sort(function(a,b){if(b.avg!==a.avg)return b.avg-a.avg;if(b.count!==a.count)return b.count-a.count;return String(a.name).localeCompare(String(b.name),'ar');});
  return ranked.slice(0,n);
}

function dashboardTodayAttendance(totalStudents){
  let today='';
  try{today=typeof localDateKey==='function'?localDateKey(new Date()):(new Date().getFullYear()+'-'+String(new Date().getMonth()+1).padStart(2,'0')+'-'+String(new Date().getDate()).padStart(2,'0'));}catch(e){today='';}
  const recs=(typeof ATTENDANCE_STATE!=='undefined'&&Array.isArray(ATTENDANCE_STATE.records))?ATTENDANCE_STATE.records:[];
  const absent={},leave={};
  recs.forEach(function(r){
    if(r.date!==today)return;
    if(r.status==='غائب')absent[r.studentId]=1;
    else if(r.status==='مجاز'||r.status==='مجاز بعذر')leave[r.studentId]=1;
  });
  const absentN=Object.keys(absent).length,leaveN=Object.keys(leave).length;
  return {absent:absentN,leave:leaveN,present:Math.max(0,(totalStudents||0)-absentN-leaveN)};
}

function backupImportJson(input){
  const file=input&&input.files&&input.files[0];
  if(input) input.value='';
  if(!file)return;
  if(!confirm('سيتم استبدال بيانات هذا الجهاز بالنسخة المختارة. متابعة؟'))return;
  const reader=new FileReader();
  reader.onload=function(){
    try{
      const pack=JSON.parse(reader.result);
      const store=pack&&pack.store?pack.store:pack;
      if(!store||typeof store!=='object'){alert('ملف النسخة غير صالح.');return;}
      Object.keys(store).forEach(function(k){
        if(k.indexOf('schoolResults')===0 && store[k]!=null) localStorage.setItem(k, typeof store[k]==='string'?store[k]:JSON.stringify(store[k]));
      });
      alert('تم استيراد النسخة. سيتم إعادة تحميل الصفحة.');
      location.reload();
    }catch(e){alert('تعذر قراءة ملف النسخة.');}
  };
  reader.readAsText(file,'utf-8');
}

function backupSectionSheetName(grade,label){
  return (String(grade||'')+' - '+String(label||'')).replace(/[\\\/\*\?\:\[\]]/g,'-');
}
function backupFindSection(sheetName){
  let raw=String(sheetName||'').trim();
  try{ if(window.nbUnaliasGrade) raw=window.nbUnaliasGrade(raw); }catch(e){}
  function matchPair(g,lab){
    const sec=(gbSectionsForGrade(g)||[]).find(function(s){return String(s.label)===String(lab);});
    return sec?{grade:g,sec:sec}:null;
  }
  const parts=raw.split('__');
  if(parts.length>=2){ const hit=matchPair(parts[0],parts.slice(1).join('__')); if(hit) return hit; }
  const dash=raw.match(/^(.+?)\s*[-–—_]\s*(\S+)$/);
  if(dash){ const hit=matchPair(dash[1].trim(),dash[2].trim()); if(hit) return hit; }
  let found=null;
  gradebookGradeLabels().forEach(function(g){
    gbSectionsForGrade(g).forEach(function(sec){
      if(found)return;
      const lab=String(sec.label||'');
      if(raw===g+' '+lab || raw===g+lab) found={grade:g,sec:sec};
    });
  });
  return found;
}
function backupExportStudentsExcel(){
  if(typeof XLSX==='undefined'){alert('مكتبة الإكسل غير متاحة.');return;}
  const wb=XLSX.utils.book_new();
  const head=['ت','اسم الطالب','اللقب','اسم الأم','تاريخ الميلاد','هاتف ولي الأمر','الحالة','محل الولادة','رقم البطاقة الوطنية','الرعاية'];
  let sheets=0;
  gradebookGradeLabels().forEach(function(g){
    gbSectionsForGrade(g).forEach(function(sec){
      const aoa=[head];
      (sec.students||[]).forEach(function(st,i){
        aoa.push([i+1,st.name||'',st.title||'',st.motherName||'',st.birthDate||'',st.guardianPhone||'',st.status||'',st.birthPlace||'',st.nationalId||'',st.careCoverage||'']);
      });
      const ws=XLSX.utils.aoa_to_sheet(aoa);
      let name=backupSectionSheetName(g,sec.label);
      if(name.length>31) name=name.slice(0,31);
      try{XLSX.utils.book_append_sheet(wb,ws,name);sheets++;}catch(e){}
    });
  });
  if(!sheets){alert('لا توجد شعب لتصديرها.');return;}
  xlsxWriteStyled(wb,'طلاب-حسب-الشعب.xlsx');
}
function backupImportStudentsExcel(input){
  const file=input&&input.files&&input.files[0];
  if(input) input.value='';
  if(!file)return;
  if(typeof XLSX==='undefined'){alert('مكتبة الإكسل غير متاحة.');return;}
  const reader=new FileReader();
  reader.onload=function(ev){
    try{
      const wb=XLSX.read(ev.target.result,{type:'array'});
      let added=0, sheets=0;
      wb.SheetNames.forEach(function(sn){
        const found=backupFindSection(sn);
        if(!found){return;}
        sheets++;
        const rows=XLSX.utils.sheet_to_json(wb.Sheets[sn],{header:1,raw:false,defval:''});
        added+=ingestStudentGrid(rows, found.sec);
      });
      if(!sheets){alert('لم يُعثر على شيتات مطابقة لأسماء الشعب. صدّر ملفاً أولاً لمعرفة التسمية.');return;}
      saveApp();
      alert('تم استيراد '+added+' طالباً من '+sheets+' شعبة.');
    }catch(e){alert('تعذر استيراد ملف الطلاب.');}
  };
  reader.readAsArrayBuffer(file);
}

function gbNormGradeLabel(x){
  return String(x==null?'':x).replace(/^\s*الصف\s+/,'').replace(/[أإآ]/g,'ا').replace(/ة/g,'ه').replace(/ى/g,'ي').replace(/\s+/g,' ').trim();
}
function gbSectionsForGrade(label){
  let grade = (APP.grades || []).find(function(g){ return g.label === label; });
  if(!grade){
    const want=gbNormGradeLabel(label);
    grade = (APP.grades || []).find(function(g){ return gbNormGradeLabel(g.label) === want; });
  }
  return grade ? (grade.sections || []) : [];
}
